	void PM_PhaseStart()
	{
	 phase = true;
	 bNOPAIN = true;
	 for(int i=2;i>0;i--)
		A_SpawnProjectile("AbhorBallExplode",random(10,45),0,random(0,360),2,random(0,360));
	 A_SetRenderStyle(0.01,STYLE_AddStencil);
	 A_StartSound("Harvester/PhaseStart",10);
	 A_StartSound("Harvester/PhaseLoop",11,CHANF_LOOPING,volume:0.85);
	}
	
	void PM_PhaseStop()
	{
	 phase = false;
	 bNOPAIN = false;
	 alpha = 1;
	 A_SetRenderStyle(1.0,STYLE_Normal);
	 for(int i=2;i>0;i--)
		A_SpawnProjectile("AbhorBallExplode",random(10,45),0,random(0,360),2,random(0,360));
	 A_StopSound(11);
	 A_StartSound("Harvester/PhaseEnd",12);
	}

	Phase:

			PM_PhaseStart();

			else A_Chase(null,null,CHF_NOPLAYACTIVE);
		}
		C12_ B 15
		{
			bFRIGHTENED = false;
			PM_PhaseStop();
			A_Stop();
			A_SetSpeed(12);
		}
		Goto See;
	Missile:
		TNT1 A 0 A_Jump(100, "SummonSpike", "Dodge");
	BoneShards:
		C12_ EEE 2 A_FaceTarget();
		C12_ FFFFGGGGGG 2
		{

		{

			A_SpawnProjectile("PM_CacolichBoneShard", 25, 0, frandom(-15, 15), CMF_OFFSETPITCH|CMF_AIMDIRECTION, frandom(-10, 10));

		}
		Loop;
	DeathEnd:

Class PM_CacolichSpike : Actor
{
	float rand;
	bool reversed;
	Default
	{
		+FLOORHUGGER
		-SOLID
		+NOINTERACTION
		+THRUACTORS
		Scale 0.1;
	}
	
	override void PostBeginPlay()
	{
		super.PostBeginPlay();
		if (!random(0, 1))
		{
			scale.x *= -1;
			reversed = true;
		}
	}
	
	States
	{
	Spawn:
		TNT1 A random(0, 6) NoDelay;
		C0S_ AAAAAAAA 1
		{
			rand = frandom(0.06, 0.15);
			scale += ((reversed) ? -rand : rand, rand);
		}
		C0S_ A random(40, 60);
		Goto Fade;
	Fade:
		C0S_ A 1
		{
			if (scale.y <= 0.0001) SetStateLabel("Nullify");
			scale -= ((reversed) ? -0.05 : 0.05, 0.05);
		}
		Loop;
	Nullify:
		TNT1 A 0;
		Stop;
	}
}
