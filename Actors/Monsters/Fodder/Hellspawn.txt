Class PM_HellspawnScreamer : PMUmbralMonsterBase
{
    Default
	{
		PMMonsterBase.CodexLore
		"Hellspawn are cobbled beings together en masse that have a single minded killing instinct, nothing more, nothing less.
		
		They are far more agressive than any melee residing lesser demons, can crawl on ceilings, climb on walls, and can leap at extremely far distances. Stay far away from them as possible.
		If one of them unleashes a prolonged scream, then be prepared for a large armada of Hellspawns and shoot every one of them until they die.
		
		You can stop the screamer by killing it before it finishes its cry.";
		Health 80;
		GibHealth 20;
		Radius 18;
		Height 56;
		Speed 15;
		PainChance 256;
		SeeSound "";
		PainSound "PMHellspawn/Pain";
		DeathSound "PMHellspawn/Death";
		ActiveSound "PMHellspawn/Idle";
		Obituary "%o faced lethal lacerations due to a Hellspawn.";
		DropItem "Clip", 50;
		Species "Fodder";
		Tag "Hellspawn";
		Monster;
		BloodColor "Black";
		Bloodtype "PM_BlackBloodBase";
		MeleeRange 60;
		XScale 0.84;
		YScale 0.91;
		+FLOORCLIP
		+NOINFIGHTING
		-Countkill
		+THRUSPECIES
		+SLIDESONWALLS
		+DONTHARMCLASS
		+DONTHARMSPECIES
		+NOBLOCKMONST
		-ACTIVATEMCROSS
		-CANUSEWALLS
		-CANPUSHWALLS
		TeleFogSourceType "PM_SmallDemonicTeleportFog";
		TeleFogDestType "PM_SmallDemonicTeleportFog";
	}
	
	void A_HellspawnChase()
	{
		A_Recoil(-frandom(0.2, 0.5));
		A_Chase();
	}
	
	bool screamed;
	
	States
	{
	Spawn:
		HAAA B 0 NoDelay;
	Idle:
		HEL_ AB 3 A_Wander();
		HEL_ A 0 A_StartSound("Horrorspawn/Step",6);
		HEL_ AB 0 A_Look();
		HEL_ CD 7 A_Wander();
		HEL_ A 0 A_StartSound("Horrorspawn/Step",6);
		HEL_ AB 0 A_Look();
		HEL_ A 0 A_Jump(5,"IdleStill");
		Loop;
	IdleStill:
		HEL_ AAAAA 10 A_Look();
		HEL_ A 0 A_Jump(1,"Idle");
		Loop;
	See:
		HEL_ A 0 A_StartSound("Horrorspawn/Step",6);
		HEL_ AABB 2 A_HellspawnChase();
		HEL_ A 0 A_StartSound("Horrorspawn/Step",6);
		HEL_ CCDD 2 A_HellspawnChase();
		Loop;
	Dash:
		HEL_ AABBCCDD 1
		{
			if (target && distance3D(target) < 200)
				SetStateLabel("See");
			else
			{
				A_Recoil(-1);
				A_FaceTarget(12,15);
				A_SpawnItemEx("PM_GenericPhaseGhost",flags:SXF_TRANSFERSTENCILCOL|SXF_TRANSFERALPHA|SXF_TRANSFERRENDERSTYLE|SXF_NOCHECKPOSITION|SXF_TRANSFERSPRITEFRAME);
				for (int i = 0; i < random(2, 3); i++) A_Chase(null,null);
			}
		}
		HEL_ A 0 A_Jump(10, "See");
		Loop;
	Missile:
		TNT1 A 0
		{ if (screamed) SetStateLabel("Dash"); }
		Goto Scream;
	Scream:
		HAAA A 0
		{
			bNOPAIN = true;
			screamed = true;
		}
		HEL_ A 10 A_FaceTarget();
		HAAA A 30 A_FaceTarget();
		HAAA A 0 A_StartSound("PMHellspawn/Call", CHAN_VOICE);
		HAAA BCDBCDBCDBCDBCEBCDEEBDCEBDCEBDCEBDCBEDBCEDBCEBDCEBDCBEDCBEDCBEDCBEDBCEDBCEDBCEBDCEDBCEBDCEBCBEDCBEDBEDCBDECBEDCEBDCEDC 1
		{
			BlockThingsIterator ti = BlockThingsIterator.Create(self, 400);
			actor a;
			
			while(ti.Next())
			{
				a = ti.thing;
				double rad = abs((400 - Distance3D(a))/2);
			}
			
			A_QuakeEx(10, 10, 0, 8, 0, 1000, "", QF_3D|QF_SCALEDOWN);
		}
		HAAA C 0
		{
			for(int i=random(15,22);i>0;i--)
				A_SpawnItemEx("PM_HellspawnHordeSpawner");
			bNOPAIN = false;
		}
		Goto See;
	Pain:
		HEL_ H 1;
		HEL_ H 1 A_Pain();
		HEL_ H 0 A_Jump(128,"Dash");
		Goto See;
	Death:
		H_EL A 6;
		H_EL B 6 A_Scream();
		H_EL C 5 A_NoBlocking();
		H_EL D 5;
		TNT1 A 0 A_StartSound("GBloodSplashBig",CHAN_6);
		TNT1 AA 0 A_SpawnItemEx("PM_TumorBloodMist",0,0,3,0,0,0,0,SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		TNT1 AAAAAAAAAA 0 A_SpawnItemEx("RedGroundOoze",random(-7,7),random(-7,7),1,random(-5,5),random(-5,5),random(-0,3),0,SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		TNT1 AAAAAAAAAAA 0 A_SpawnItemEX("PM_TaintedDeathSplash", 0, 0, 0, random (1, 3), random(1,3), random (1, 6), random(0, 360),SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		TNT1 AAAAAAAAAAAAAAA 0 A_SpawnItemEX("PM_TumorBurstBlood", 0, 0, 0, random (-15, 15), random(-15,15), random (1, 7),0,SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		H_EL EEEFFFFFFFF 4 A_SpawnItemEx("PM_TumorBurstBlood",random(-40,40),random(-40,40),0,random(-4,4),random(-4,4),random(0,5),0,SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		H_EL F -1;
		Stop;
	}
}

Class PM_HellspawnHordeSpawner : PM_AfflictedHordeSpawner
{
	States
	{
	Spawn:
		TNT1 A 0 NoDelay { bFRIGHTENED = true; }
		TNT1 A 0 A_Wander();
		TNT1 AAAAA random(0,10) A_Chase();
		TNT1 A 0 A_Jump(30,"SpawnLoop");
		Loop;
	SpawnLoop:
		TNT1 A 0 { bFRIGHTENED = false; }
		TNT1 A 5 A_Wander();
		TNT1 A 0 A_CheckSight ("Create");
		TNT1 A 5 A_Wander();
		TNT1 A 0 A_CheckSight ("Create");
		TNT1 A 0 A_Jump(15,"Die");
		Loop;
	Create:
		TNT1 A 0 A_SpawnItemEx("PM_Hellspawn");
		TNT1 A 0 A_StartSound("PMHellspawn/Horde",38,pitch:frandom(0.8,1.3));
		Stop;
	Die:
		Stop;
	}
}

Class PM_Hellspawn : PMUmbralMonsterBase
{
	bool grouped;
	int oldspeed;
	
    Default
	{
		Health 50;
		GibHealth 20;
		Radius 18;
		Height 56;
		Speed 10;
		PainChance 256;
		SeeSound "PMHellspawn/Roar";
		PainSound "PMHellspawn/Pain";
		DeathSound "PMHellspawn/Death";
		ActiveSound "PMHellspawn/Idle";
		Obituary "%o faced lethal lacerations due to a Hellspawn.";
		DropItem "Clip", 50;
		Species "Fodder";
		Tag "Hellspawn";
		Monster;
		BloodColor "Black";
		Bloodtype "PM_BlackBloodBase";
		MeleeRange 60;
		XScale 0.84;
		YScale 0.91;
		+FLOORCLIP
		+NOINFIGHTING
		-Countkill
		+THRUSPECIES
		+SLIDESONWALLS
		+DONTHARMCLASS
		+DONTHARMSPECIES
		+NOBLOCKMONST
		-ACTIVATEMCROSS
		-CANUSEWALLS
		-CANPUSHWALLS
		TeleFogSourceType "PM_SmallDemonicTeleportFog";
		TeleFogDestType "PM_SmallDemonicTeleportFog";
	}
	
	void A_HellspawnChase()
	{
		if (grouped) A_Recoil(-frandom(0.2, 0.8));
		else A_Recoil(-frandom(0.1, 0.3));
		A_Chase();
	}
	
	override void PostBeginPlay()
	{
		super.PostBeginPlay();
		oldspeed = int(ceil(speed));
	}
	
	States
	{
	Spawn:
		HAAA B 0 NoDelay;
	Idle:
		HEL_ AB 3 A_Wander();
		HEL_ A 0 A_StartSound("Horrorspawn/Step",6);
		HEL_ AB 0 A_Look();
		HEL_ CD 7 A_Wander();
		HEL_ A 0 A_StartSound("Horrorspawn/Step",6);
		HEL_ AB 0 A_Look();
		HEL_ A 0 A_Jump(5,"IdleStill");
		Loop;
	IdleStill:
		HEL_ AAAAA 10 A_Look();
		HEL_ A 0 A_Jump(1,"Idle");
		Loop;
	See:
		HEL_ A 0 
		{
			if(CheckProximity("PM_Hellspawn",450,8))
			{
				grouped = true;
				speed = random(10,18);
			}
			else
			{
				grouped = false; 
				speed = oldspeed;
			}
		}
		HEL_ A 0 A_StartSound("Horrorspawn/Step",6);
		HEL_ AABB 2 A_HellspawnChase();
		HEL_ A 0 A_StartSound("Horrorspawn/Step",6);
		HEL_ CCDD 2 A_HellspawnChase();
		Loop;
	Melee:
		TNT1 A 0 A_Jump(40, "Retreat");
		HEL_ E 0
		{
			A_StartSound("PMHellspawn/Attack", CHAN_VOICE);
			A_StartSound("Skulker/Miss", CHAN_BODY);
			A_Recoil(-10);
			vel.z += frandom(2, 4);
			A_FaceTarget();
		}
		HEL_ EF 3 A_FaceTarget();
		HEL_ G 3 A_CustomMeleeAttack(random(6,12),"ZombieFodder/Melee");
		HEL_ F 3 A_FaceTarget();
		Goto See;
	Retreat:
		HEL_ F 15
		{
			A_StartSound("PMHellspawn/Pain", CHAN_VOICE);
			A_Recoil(20);
			vel.z += frandom(3, 8);
		}
		Goto See;
	Dash:
		HEL_ AABBCCDD 1
		{
			if (target && distance3D(target) < 200)
				SetStateLabel("Melee");
			else
			{
				A_Recoil(-1);
				A_FaceTarget(12,15);
				A_SpawnItemEx("PM_GenericPhaseGhost",flags:SXF_TRANSFERSTENCILCOL|SXF_TRANSFERALPHA|SXF_TRANSFERRENDERSTYLE|SXF_NOCHECKPOSITION|SXF_TRANSFERSPRITEFRAME);
				for (int i = 0; i < random(2, 3); i++) A_Chase("Melee",null);
			}
		}
		HEL_ A 0 A_Jump(4, "See", "Leap");
		Loop;
	Missile:
		TNT1 A 0 A_Jump(100, "Leap");
		Goto Dash;
	Leap:
		TNT1 A 0 A_Jump(40,"See");
	    TNT1 A 0 
	    { 
	     if(target && Distance3D(target) >= 1680)
	      { SetStateLabel("See"); }
	    }
		HEL_ GGFF 4  A_FaceTarget();
		HEL_ E 2 PM_LeapTo(target, 1680, "Dash");
		TNT1 A 0 A_StartSound("Squire/Leap",15);
		TNT1 A 0 A_Recoil(-4);
	LeapLoop:
		HEL_ EEEE 1 
		{
		 if(random(1,10) == 1)
		  { ThrustThing(random(0,360)*256/360,2,0,0); }
		  A_SpawnItemEx("PM_GenericPhaseGhost",flags:SXF_TRANSFERSTENCILCOL|SXF_TRANSFERALPHA|SXF_TRANSFERRENDERSTYLE|SXF_NOCHECKPOSITION|SXF_TRANSFERSPRITEFRAME);
		}
        TNT1 A 0 A_CheckFloor("LeapEnd");
		Loop;
	LeapEnd:
		HAAA A 10 A_Stop();
		Goto See; 
	Pain:
		HEL_ H 1;
		HEL_ H 1 A_Pain();
		HEL_ H 0 A_Jump(128,"Dash","Melee");
		Goto See;
	Death:
		H_EL A 6;
		H_EL B 6 A_Scream();
		H_EL C 5 A_NoBlocking();
		H_EL D 5;
		TNT1 A 0 A_StartSound("GBloodSplashBig",CHAN_6);
		TNT1 A 0 A_SpawnItemEx("PM_TumorBloodMist",0,0,3,0,0,0,0,SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		TNT1 AAAAAAAAAA 0 A_SpawnItemEx("RedGroundOoze",random(-7,7),random(-7,7),1,random(-5,5),random(-5,5),random(-0,3),0,SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		TNT1 AAAAAAAAAAA 0 A_SpawnItemEX("PM_TaintedDeathSplash", 0, 0, 0, random (1, 3), random(1,3), random (1, 6), random(0, 360),SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		TNT1 AAAAAAAAAAAAAAA 0 A_SpawnItemEX("PM_TumorBurstBlood", 0, 0, 0, random (-15, 15), random(-15,15), random (1, 7),0,SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		H_EL EEEFFFFFFFF 4 A_SpawnItemEx("PM_TumorBurstBlood",random(-40,40),random(-40,40),0,random(-4,4),random(-4,4),random(0,5),0,SXF_NOCHECKPOSITION|SXF_USEBLOODCOLOR);
		H_EL P -1;
		Stop;
	}
}
