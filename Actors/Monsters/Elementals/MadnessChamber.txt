Class PM_MadnessChamber : PMWrathMonsterBase
{
	Default
	{
		PMMonsterBase.CodexLore
		"Packs of Furors sometimes tend to get inside spherical cages, in which attracts more of them to the point where they go into a state of madness.
		Those souls will tend to charge out of the cage to cause destruction around the area, and sometimes aim it towards hostiles.";
		Health 330;
		PainChance 50;
		Speed 3;
		Radius 31;
		Height 56;
		Mass 500;
		Monster;
		SeeSound  "MadnessChamber/Idle";
		PainSound "MadnessChamber/Pain";
		DeathSound "MadnessChamber/Death";
		ActiveSound "MadnessChamber/Idle";
		Renderstyle "Normal";
		Obituary "%o couldn't handle the madness.";
		Tag "Madness Chamber";
		Species "Elemental";
		//+NOINFIGHTING
		+FLOATBOB
		+FLOAT
		FloatBobStrength 0.1;
		+NOGRAVITY
		+NOICEDEATH
		+DONTFALL
		+NOBLOOD
		+FORCEXYBILLBOARD
		+DONTHARMCLASS
		+NOFEAR
		+DONTHARMSPECIES
		DamageFactor "SoulExplode", 0.0;
		TeleFogSourceType "PM_SmallDemonicTeleportFog";
		TeleFogDestType "PM_WrathTeleportFog";
	}
	
	override void Tick()
	{
		super.Tick();
		if (target) vel *= 0.925;
	}
	
	override void PostBeginPlay()
	{
	 tex[0] = TexMan.CheckForTexture("GNSML0",TexMan.Type_Sprite);
	 tex[1] = TexMan.CheckForTexture("SPKOA0",TexMan.Type_Sprite);
	 tex[2] = TexMan.CheckForTexture("LEYSO0",TexMan.Type_Sprite);
	 tex[3] = TexMan.CheckForTexture("LENRA0",TexMan.Type_Sprite);
	 tex[4] = TexMan.CheckForTexture("BLHDA0",TexMan.Type_Sprite);
	 tex[5] = TexMan.CheckForTexture("DUSTC0",TexMan.Type_Sprite);
	 tex[6] = TexMan.CheckForTexture("GNSMK0",TexMan.Type_Sprite);
	}
	
	TextureID tex[7];
	
	void ErraticFire()
	{
		if (!random(0, 1)) A_SpawnProjectile("PM_FurorMissile",24,angle:frandom(0,360),flags:CMF_OFFSETPITCH,pitch:frandom(0,360));
	}
	
	States
	{
	Spawn:
		E04_ ABC 5 PM_Look();
		Loop;
	Look:
		E04_ AAAAABBBBBCCCCC 1
		{
		vel.x += frandom(-1, 1);
		vel.y += frandom(-1, 1);
		vel.z += frandom(-1, 1);
		PM_LookingForPlayer();
		if (target && CheckSight(target) && (level.maptime % random(8,10)) == 0) ErraticFire();
		}
		TNT1 A 0 A_JumpIfTargetInLOS("Spotted",270,JLOSF_DEADNOJUMP);
		Loop;
	Spotted:
		TNT1 A 0 PM_AlertSound();
		Goto See;
	See:
		E04_ A 0 A_ChangeVelocity(random(-1,1), random(-1,1), random(-1,1));
		E04_ AAAAA 1
		{
		vel.x += frandom(-1, 1);
		vel.y += frandom(-1, 1);
		vel.z += frandom(-1, 1);
		A_Chase(flags:CHF_DONTMOVE);
		if ((level.maptime % random(8,10)) == 0) ErraticFire();
		}
		Loop;
	Missile:
		E04_ AAAAADDDDDEEEEE 1
		{
		A_FaceTarget();
		if ((level.maptime % random(8,10)) == 0) ErraticFire();
		}
		E04_ F 5
		{
		for (int i = 0; i < 2; i++)
			A_SpawnProjectile("PM_FurorCharge",24,angle:frandom(-20, 20),flags:CMF_OFFSETPITCH|CMF_AIMDIRECTION,pitch:frandom(-20,20));
		}
		E04_ EDA 5 ErraticFire();
		Goto See;
	Pain:
		E04_ G 3;
		E04_ G 3 A_Pain();
		Goto See;
	Death:
		E04_ H 5;
		E04_ I 6 PM_Scream();
		E04_ JJKKLLM 3 Light("AvatarFireball4")
		{
		  A_SpawnParticleEx("",tex[3],style: STYLE_Add,flags: SPF_RELATIVE|SPF_FULLBRIGHT|SPF_ROLL,lifetime: random(3,6),size: frandom(280.0,450.0),xoff: 5,startalphaf: frandom(0.8,1),startroll: random(-180,180));
		  A_SpawnItemEx("PM_BigFireGlobs",random(-20,20),random(-20,20),random(-20,20),random(-5,5),random(-5,5),random(-5,5),random(0,360),SXF_NOCHECKPOSITION);
		}
		TNT1 A 0 Radius_Quake(4, 24, 0, 5, 0);
		TNT1 A 0 A_NoBlocking();
		TNT1 A 0
		{
		 for(int i=random(15,20);i>0;i--)
		  A_SpawnParticleEx("727272",tex[5],style: STYLE_Normal,flags: SPF_RELATIVE|SPF_ROLL,lifetime: random(40,70),size: frandom(2,35),zoff: random(10,60),velx: frandom(-8,8),vely: frandom(-8,8),velz: frandom(-8,12),accelz: frandom(-0.1,-0.5),startalphaf: 1.0,sizestep: -0.2,startroll: random(-180,180),rollvel: random(-4,4));
		}
		TNT1 A 0 A_Explode(60,200,XF_NOTMISSILE,0,15,0,0,"BulletPuff","Fire");
		TNT1 AAAAAAAAA 0
		{
		  A_SpawnItemEx("PM_LavaMeatChunkLauncher",random(-8,8),random(-8,8),random(5,16),random(-8,8),random(-8,8),random(0,8),0,SXF_NOCHECKPOSITION);
		  A_SpawnItemEx("PM_InfernalLavaSplashBig",random(-8,8),random(-8,8),random(-2,8),random(-8,8),random(-8,8),random(-2,8),0,SXF_NOCHECKPOSITION);
		}
		TNT1 AA 0 A_SpawnItemEx("PM_FurorNoKill",random(-10,10),random(-10,10),random(-10,10)+24,frandom(-5,5),frandom(-5,5),frandom(-2,2),frandom(0,360),SXF_NOCHECKPOSITION);
		E04_ Q 1
		{
			A_StartSound("Noble/BombExpFar", CHAN_6);
			A_StartSound("PMInfernal/Explode", CHAN_7);
			
			for (int i=5;i>0;i--) A_SpawnProjectile("PM_DarkExpSmoke",7,0,random(0,360),2,random(0,360));
			for (int i=25;i>0;i--) A_SpawnItemEx("PM_BigFireGlobs",0,0,0,random(-11,11),random(-11,11),random(-9,11),0,SXF_NOCHECKPOSITION);
			for (int i=5;i>0;i--)
			{
				A_SpawnItemEx("PM_ExplosionFlareSpawner",0,0,0,0,0,0,0,SXF_NOCHECKPOSITION,0);
				A_SpawnItemEx("PM_GenericExplosionFlames",0,0,0,random(-3,3),random(-3,3),random(-2,3),0,SXF_NOCHECKPOSITION);
				A_SpawnItemEx("PM_VoidSmoke",0,0,20,random(-1,1),random(-1,1),random(-1,1),random(1,360));
			}
		}
		Stop;
	}
}

Class PM_FurorCharge : DoomImpBall
{
	Default
	{
		DamageFunction (35);
		Speed 22;
		Radius 16;
		Height 25;
		Projectile;
		SeeSound "Furor/Scream";
		DeathSound "Furor/Death";
		DamageType "SoulExplode";
		RenderStyle "Normal";
		Obituary "%o got bombarded by a host of Furor.";
		+NEVERFAST
		+SEEKERMISSILE
		+THRUSPECIES
		Species "Soul";
	}
	
	override void Tick()
	{
		if (level.isFrozen()) return;
		super.Tick();
		
		if ((level.maptime % 3) == 0)
		{
			if (!random(0, 1)) A_SpawnItemEx("PM_InfernalRushFire",-7+frandom(-2,2),frandom(-4,4),frandom(-5,5)+32,frandom(-1,1),frandom(-1,1),frandom(-1,1),flags:SXF_NOCHECKPOSITION);
			else A_SpawnItemEx("PM_ExileFireGlobs",-7+frandom(-2,2),frandom(-4,4),frandom(-5,5)+32,frandom(-1,1),frandom(-1,1),frandom(-1,1),flags:SXF_NOCHECKPOSITION);
			if ((level.maptime % 2) == 0) A_SpawnItemEx("PM_HuskFireGlobs",-7+frandom(-7,7),frandom(-7,7),frandom(-7,7)+32,frandom(-1,1),frandom(-1,1),frandom(-1,1),flags:SXF_NOCHECKPOSITION);
		}
	}
	
	States
	{
	Spawn:
		S04_ C 1 A_Tracer2(1.1);
		Loop;
	Death:
		S04_ DE 3;
		S04_ F 3
		{
			A_StartSound("Noble/BombExp",9,pitch:1.3);
			for (int i = 0; i < 3; i++)
			{
				A_SpawnProjectile("PM_SmallBurntSoulPart1",0,0,random(0,360),2,random(0,360));
				A_SpawnProjectile("PM_SmallBurntSoulPart3",0,0,random(0,360),2,random(0,360));
				A_SpawnProjectile("PM_DarkExpSmoke",7,0,random(0,360),2,random(0,360));
				A_SpawnProjectile("PM_HectebusBallExplode",0,0,random(0,360),2,random(0,360));
				A_SpawnItemEx("PM_BigFireGlobs",0,0,0,random(-11,11),random(-11,11),random(-9,11),0,SXF_NOCHECKPOSITION);
			}
			A_SpawnItemEx("PM_ExplosionFlareSpawner",0,0,0,0,0,0,0,SXF_NOCHECKPOSITION,0);
		}
		Stop;
	}
}

Class PM_FurorMissile : PM_FurorCharge
{
	Default
	{
		DamageFunction (25);
		Speed 20;
		Obituary "%o couldn't handle the madness.";
		-SEEKERMISSILE;
		Scale 0.7;
		SeeSound "";
	}
	
	override void Tick()
	{
		if (level.isFrozen()) return;
		DoomImpBall.Tick();
		
		if ((level.maptime % 2) == 0) A_SpawnItemEx("PM_HuskFireGlobs",-7+frandom(-2,2),frandom(-2,2),frandom(-2,2)+32,frandom(-1,1),frandom(-1,1),frandom(-1,1),flags:SXF_NOCHECKPOSITION);
	}
	
	States
	{
	Spawn:
		S04_ C 1 A_Tracer2(1.1);
		Loop;
	Death:
		S04_ DE 3;
		S04_ F 3
		{
			A_StartSound("Noble/BombExp",9,pitch:1.6);
			for (int i = 0; i < 4; i++) A_SpawnItemEx("PM_BigFireGlobs",0,0,0,random(3,3),random(3,3),random(-3,3),0,SXF_NOCHECKPOSITION);
			A_SpawnItemEx("PM_ExplosionFlareSpawner",0,0,0,0,0,0,0,SXF_NOCHECKPOSITION,0);
		}
		Stop;
	}
}
