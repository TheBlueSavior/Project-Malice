	override void PostBeginPlay()

		A_FaceTarget(1.5,1);
		A_Recoil(-frandom(3, 3.5));
	}
		
    States
    {
    Spawn:
		B02_ AAAAAAAAAA 10 PM_Look();

		{
			if (!random(0, 30))
				A_StartSound(activesound,554,volume:frandom(0.05,0.3),pitch:frandom(0.60,0.78));
		}

			if (!random(0, 12))

		}

		TNT1 A 0 A_StartSound("ProfaneOne/Step",4,volume:0.4);
		B02_ AAAABBBB 4
		{
		PM_Look();
		A_Wander();
		}
		TNT1 A 0 A_StartSound("ProfaneOne/Step",4,volume:0.4);
		B02_ CCCCDDDD 4

		A_Wander();
		PM_Look();
		}

		Loop;

        B02_ AAABBB 3 PM_LookingForPlayer();
		TNT1 A 0
		{
			A_Recoil(-1.5);
			A_Quake(1, 4, 0, 312);
			A_StartSound("ProfaneOne/Step",3);
		}
        B02_ B 3 A_JumpIfTargetinLOS("Spotted",250,JLOSF_DEADNOJUMP);

		}

		}

		}
		B02D L 0 A_StartSound(SeeSound, CHAN_VOICE);
		B02D MNONMNONM 4 A_FaceTarget();
		B02D M 0
		{
			bNOPAIN = true;
			bNODROPOFF = true;
		}
		Goto ChargeLoop;
	ChargeLoop:
		B02D PP random(1,2) ProfanedCharge(); 
		B02D P 0 A_StartSound("ProfaneOne/Step",3);
		TNT1 A 0 A_JumpIf(sqrt(vel.x * vel.x + vel.y * vel.y) < 3, "ChargeImpact");

		B02D P 0 A_StartSound("ProfaneOne/Step",3);
		TNT1 A 0 A_JumpIf(sqrt(vel.x * vel.x + vel.y * vel.y) < 3, "ChargeImpact");

		TNT1 A 0 A_JumpIf(sqrt(vel.x * vel.x + vel.y * vel.y) < 3, "ChargeImpact");
		B02D SS random(1,2) ProfanedCharge();
