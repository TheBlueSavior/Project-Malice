Class PM_EntrailDreg : PMGluttonyMonsterBase
{
  Default
  {
  PMMonsterBase.CodexLore
  "Fleshy carcasses that are possessed by Gluttony sprites and further mutated due to the effects of hell.
  They are equipped with metallic armor and two shoulder laser cannons for combat. At times, they might spew out regurgitated balls of flesh that home in the player.
  They are tankier than the average Cadaver, but also slightly more slower.";
  Health 420;
  Radius 20;
  Height 56;
  Mass 500;
  Speed 7;
  PainChance 80;
  MeleeThreshold 160;
  Monster;
  +FLOORCLIP
  +DONTHARMCLASS
  +DONTHARMSPECIES
  +DONTHARMCLASS
  +MISSILEMORE 
  Species "Revenant";
  SeeSound "EntrailDreg/Sight";
  PainSound "EntrailDreg/Pain";
  DeathSound "EntrailDreg/Death";
  ActiveSound "EntrailDreg/Idle";
  HitObituary "%o has been dismissed by the Militant.";
  Obituary "%o was put to a halt by the Militant.";
  Tag "Entrail Dreg";
  Bloodtype "PM_NormalCyberBloodBase";
  TeleFogSourceType "PM_SmallDemonicTeleportFog";
  TeleFogDestType "PM_GluttonyTeleportFog";
  }
  
  States
  {
  Spawn:
	R03_ B 0 NoDelay;
  Idle:
	R03_ AAAAAAAAAA 10 PM_Look();
	R03_ A 0
	{
		if (!random(0, 30))
			A_StartSound(activesound,554,volume:frandom(0.1,0.5),pitch:frandom(0.60,0.78));
	}
	R03_ A 0 A_Jump(45,"Explore");
    Loop;
  Explore:
	TNT1 A 0
	{
		if (!random(0, 12))
			A_StartSound(activesound,554,volume:frandom(0.1,0.5),pitch:frandom(0.60,0.78));
	}
	TNT1 A 0 A_SetSpeed(3);
	TNT1 A 0 A_StartSound("skeleton/step",4,volume:0.4);
	R03_ AAABBBCCC 3
	{
	PM_Look();
	A_Wander();
	}
	TNT1 A 0 A_StartSound("skeleton/step",4,volume:0.4);
	R03_ DDDEEEFFF 3
	{
	A_Wander();
	PM_Look();
	}
	R03_ A 0 A_Jump(40,"Idle");
    Loop;
  Look:
	TNT1 A 0 A_SetSpeed(6);
	TNT1 A 0 A_JumpIfTargetInLOS("Spotted",180,JLOSF_DEADNOJUMP);
    R03_ AAABBBCCC random(2,4) PM_LookingForPlayer();
	NULL A 0 A_StartSound("Afflicted/RunStep",5,pitch:0.5);
	TNT1 A 0 A_JumpIfTargetInLOS("Spotted",180,JLOSF_DEADNOJUMP);
	R03_ DDDEEEFFF random(2,4) PM_LookingForPlayer();
	NULL A 0 A_StartSound("Afflicted/RunStep",5,pitch:0.5);
    Loop;
  Spotted:
    TNT1 A 0 A_Jump(60,"SpotMissile");
    Goto Roar;
  Roar:
    TNT1 A 0 PM_AlertSound();
    Goto See;
  SpotMissile:
    TNT1 A 0 PM_AlertSound();
    R03_ F 10 A_FaceTarget;
    Goto Missile;
  See:
	TNT1 A 0 A_SetSpeed(7);
    R03_ AAABBBCCC random(2,4) A_Chase();
	NULL A 0 A_StartSound("Afflicted/RunStep",5,pitch:0.5);
	R03_ DDDEEEFFF random(2,4) A_Chase();
	NULL A 0 A_StartSound("Afflicted/RunStep",5,pitch:0.5);
    Loop;
  Melee:
    R03_ G 5
	{
		A_FaceTarget();
		A_StartSound("Doom4/Other/MeleeSwing",20);
	}
    R03_ H 5 A_FaceTarget();
    R03_ I 6 A_CustomMeleeAttack(35,"Doom4/Other/MeleeHit");
    Goto See;
  Missile:
	TNT1 A 0 A_Jump(100, "DualLaser", "MeatSpit", "MeatSpitMulti");
  Lasers:
	R03_ AK 9 A_FaceTarget();
	R03_ K 0 { pm_attackloop = random(2, 4); }
	Goto LasersLoop;
  LasersLoop:
	R03_ J 5
	{
		if (--pm_attackloop <= 0)
			SetStateLabel("Cooldown");
		A_FaceTarget(60, 60);
		A_StartSound("CyberZ/Fire",10,pitch:0.75);
		A_SpawnProjectile("PM_DregLaserTracer", 56, ((pm_attackloop % 2) == 0) ? 17 : -17);
	}
	Loop;
  Cooldown:
	R03_ K 20 A_FaceTarget();
	Goto See;
  DualLaser:
	R03_ A 0 A_StartSound("EntrailDreg/Attack",CHAN_VOICE);
	R03_ AK 13 A_FaceTarget();
	R03_ K 0 { pm_attackloop = randompick(1, 1, 2); }
	Goto DualLaserLoop;
  DualLaserLoop:
	R03_ J random(7, 10)
	{
		if (--pm_attackloop <= 0)
			SetStateLabel("Cooldown");
		PM_LeadTarget(15);
		A_StartSound("CyberZ/Fire",10,pitch:0.65);
		for (int i = -1; i < 2; i += 2) A_SpawnProjectile("PM_DregLaserTracer", 56, i * 17);
	}
	R03_ J 6 A_FaceTarget();
	Loop;
  MeatSpit:
	R03_ A 0 A_StartSound("EntrailDreg/Attack",CHAN_VOICE);
	R03_ AK 10 A_FaceTarget();
	R03_ K 0 A_StartSound("Afflicted/Puke",15,pitch:0.85);
	R03_ KKKKKKKK 3
	{
		A_FaceTarget();
		A_SpawnItemEx("MaliceBloodSplash",random(5,10),random(-2,2),random(-3,0)+62,random(3,6),random(-3,3),frandom(-4,4));
    }
	R03_ K 15
	{
		A_SpawnItemEx("PM_NormalMeatChunkLauncher",random(5,10),random(-2,2),random(-3,0)+62,random(3,6),random(-3,3),frandom(-4,4));
		PM_LeadTarget(20);
		A_SpawnProjectile("PM_DregMeat",62,flags:CMF_OFFSETPITCH|CMF_AIMDIRECTION);
	}
	R03_ A 10 A_FaceTarget();
	Goto See;
  MeatSpitMulti:
	R03_ A 0 A_StartSound("EntrailDreg/Attack",CHAN_VOICE);
	R03_ AK 10 A_FaceTarget();
	R03_ K 0 A_StartSound("Afflicted/Puke",15,pitch:0.85);
	R03_ KKKKKKKKKKKKKKK 2
	{
		A_FaceTarget();
		A_SpawnItemEx("MaliceBloodSplash",random(5,10),random(-2,2),random(-3,0)+62,random(3,6),random(-3,3),frandom(-4,4));
    }
	R03_ KKKKKK random(3, 7)
	{
		A_SpawnItemEx("PM_NormalBloodMulti",random(5,10),random(-2,2),random(-3,0)+62,random(3,6),random(-3,3),frandom(-4,4));
		A_FaceTarget();
		A_SpawnProjectile("PM_DregMeatSmall",62,angle:frandom(-30, 30),flags:CMF_OFFSETPITCH|CMF_AIMDIRECTION,pitch:frandom(-30, 30));
	}
	R03_ A 10 A_FaceTarget();
	Goto See;
  Pain:
    R03_ L 5;
    R03_ L 5 A_Pain;
    Goto See;
  Death:
	R03_ L 5 A_StartSound("PM/GibSplat",30,pitch:0.6);
	R03_ M 5 PM_Scream();
    R03_ NOP 5 A_SpawnItemEx("PM_NormalMeatChunkLauncher",random(-2,2),random(-2,2),random(10,50),0,0,0,0,SXF_NOCHECKPOSITION);
	R03_ Q 5
	{
		A_SpawnItemEx("PM_NormalMeatChunkLauncher",random(-2,2),random(-2,2),random(10,50),0,0,0,0,SXF_NOCHECKPOSITION);
		A_StartSound("Corpse/FallMetal",CHAN_6);
	}
    R03_ Q -1;
    Stop;
  Raise:
    R03_ Q 5;
    R03_ PONML 5;
    Goto Look;
  }
}

Class PM_DregLaserTracer : PM_WarHoundSniperTracer
{
	Default
	{
		Speed 25;
		FastSpeed 30;
		DamageFunction 15;
		DamageType "Plasma";
		SeeSound "EntrailDreg/Laser";
		+SEEKERMISSILE
	}
	
	override void PostBeginPlay()
	{
		Super.PostBeginPlay();
		tx[0] = TexMan.CheckForTexture("SPKRA0",TexMan.Type_Sprite);
		tx[1] = TexMan.CheckForTexture("DTPRR0",TexMan.Type_Sprite);
		tx[2] = TexMan.CheckForTexture("GNSMK0",TexMan.Type_Sprite);
	}
   
	TextureID tx[3];

	{
		super.Tick();

		{

		}
	}
	
	States
	{
	Spawn:

Class PM_DregMeat : PMProjectileBase
{
	Default
	{
		DamageFunction (35);
		DamageType "Blood";
		SeeSound "monster/vulsh1";
		DeathSound "GBloodSplashBig";
		Speed 6;
		FastSpeed 9;
		+SEEKERMISSILE
		+ROLLSPRITE
		+FORCEXYBILLBOARD
		RenderStyle "Normal";
		Radius 5;
		Height 5;
		Decal "Bloodsplat";
		Scale 1.55;
	}
	
	int rollp, time;
	bool launched;
	vector3 lastpos;
	
	override void PostBeginPlay()
	{
		super.PostBeginPlay();
		lastpos=self.pos;
		rollp = frandom(-4, 4);
	}
	
	override void Tick()
	{
		super.Tick();
		if (!level.isFrozen())
		{
			roll += rollp;
			if (launched)
			{
				let Coord = PM_Trailer.Trail(Self,lastpos,"ff1010",(-4,0,0),1.4,0.25,7);
				lastpos=Coord;
			}
			else lastpos=pos;
			if (!(getAge() % 3)) A_SpawnItemEx("MaliceBloodSplash",random(-2,2),random(-2,2),random(-2,2),random(3,6),random(-3,3),frandom(-4,4),flags:SXF_NOCHECKPOSITION);
		}
	}
	
	States
	{
	Spawn:
		R03M A 1
		{
			if (++time >= 70) SetStateLabel("Launch");
			A_Tracer2(1.5);
		}
		Loop;
	Launch:
		R03M AAAAAAAAA 1 { vel *= 0.9; }
		R03M A 10;
		R03M A 0
		{
			A_SpawnParticleEx("",TexMan.CheckForTexture("LENRA0",TexMan.Type_Sprite),STYLE_Add,SPF_ROLL|SPF_RELATIVE|SPF_FULLBRIGHT,random(10,20),100,sizestep:-1);
			A_SeekerMissile(999, 999);
			A_StartSound("Grievance/Launch", 10, pitch: 1.25);
			launched = true;
		}
		R03M AAAAAAA random(1,2) { vel *= 1.2; }
		Goto LaunchLoop;
	LaunchLoop:
		R03M A 1;
		Loop;
	Death:
		TNT1 AAAA 0 A_SpawnItemEx("PM_NormalBloodMulti",xvel: frandom(-1,1),yvel: frandom(-1,1),zvel: frandom(-1,1),flags: SXF_NOCHECKPOSITION|SXF_ABSOLUTEVELOCITY);
		TNT1 AAAA 0 A_SpawnItemEx("PM_NormalMeatChunkLauncher",xvel: frandom(-1,1),yvel: frandom(-1,1),zvel: frandom(-1,1),flags: SXF_NOCHECKPOSITION|SXF_ABSOLUTEVELOCITY);
		Stop;
	}
}
