	    (

			A_Explode((70),310,XF_NOTMISSILE,0,50,0,0,"BulletPuff","PM_InfernalMelee");

			(
			"",
			tex[3],
			style: STYLE_Add,
			flags: SPF_RELATIVE|SPF_FULLBRIGHT|SPF_ROLL,
			lifetime: 8,
			size: 360,
			angle: 0,
			zoff: 40,
			startalphaf: 1,
			startroll: random(-180,180)
			);
			A_SpawnParticleEx
			(   
			"",
			tex[4],
			style: STYLE_Add,
			flags: SPF_ROLL|SPF_RELATIVE|SPF_FULLBRIGHT,
			lifetime: 20,
			size: 800,
			xoff: 0,
			yoff: 0,
			zoff: 50,
			startalphaf: 1,
			fadestepf: -1,
			sizestep: -25,
			startroll: random(-180,180)
			);
			for (int i = 30; i > 0; i--)
			{
			A_SpawnParticleEx
			(
			"727272",
			tex[5],
			style: STYLE_Normal,
			flags: SPF_RELATIVE|SPF_ROLL,
			lifetime: random(40,70),
			size: frandom(2,35),
			angle: 0,
			xoff: 0,

			velx: frandom(-8,8),

			accelx: 0,
			accely: 0,
			accelz: frandom(-0.1,-0.5),
			startalphaf: 1.0,
			fadestepf: 0,
			sizestep: -0.2,
			startroll: random(-180,180),
			rollvel: random(-4,4),
			rollacc: 0
			);
			A_SpawnParticleEx
			(
			"6C6C6C",
			tex[6],
			style: STYLE_Normal,
			flags: SPF_RELATIVE|SPF_ROLL,
			lifetime: random(30,60),
			size: frandom(80,220),
			zoff: random(10,60),
			velx: random(-9,9),
			vely: random(-9,9),
			velz: random(-9,9),

	States
	{
		Spawn:
			A04_ A 10 PM_Look();
			Loop;
		Look:
			TNT1 A 0 A_StartSound("Exile/Step",CHAN_5);
			A04_ AABBCC 3 PM_LookingForPlayer();
			TNT1 A 0 A_JumpifTargetInLOS("Spotted",180,JLOSF_DEADNOJUMP);
			TNT1 A 0 A_StartSound("Exile/Step",CHAN_5);
			A04_ DDEEFF 3 PM_LookingForPlayer();
			TNT1 A 0 A_CheckSight("See");
			Goto Spotted;
		Spotted:
			TNT1 A 0 PM_AlertSound();
		See:
			TNT1 A 0
			{ if(!ire && pm_irecount <= 0) SetStateLabel("Ire"); }
			TNT1 A 0 A_SetSpeed(15);
			TNT1 A 0 A_StartSound("Exile/Step",CHAN_5);
			A04_ AABBCC 3
			{
				if (ire) tics = 2;
				A_VileChase();
			}
			TNT1 A 0 A_StartSound("Exile/Step",CHAN_5);
			A04_ DDEEFF 3
			{
				if (ire) tics = 2;
				A_VileChase();
			}
			Loop;
		Heal:
			A04B ABCB 5;
			Goto See;
		Ire:
			TNT1 A 0 A_StartSound("GenericChargeSmall",CHAN_5);
			A04B ABCB 4 A_FaceTarget();
			TNT1 A 0 
			{ 
				ire = true;
				bNOINFIGHTING = true;
				bNOPAIN = true;
				bNOFEAR = true;
				A_SpawnItemEx("PM_WrathIreActivate",0,0,20);
				A_SpawnItemEx("PM_WrathIreEffect",0,0,20,flags:SXF_SETMASTER);
			}
			TNT1 A 0 A_SetSpeed(12);
			TNT1 A 0 PM_IreStartSound();
			TNT1 A 0 A_StartSound(ActiveSound,CHAN_7,CHANF_DEFAULT,1,3);
			A04B ABCBABCB 4 A_FaceTarget();
			Goto See;
		Geysers:
			A04_ M 0
			{
				if (ire) pm_attackloop = random(4, 5);
				else pm_attackloop = random(3, 4);
				A_StartSound("Immolator/Attack", CHAN_VOICE);
			}
			A04_ GGHHIIGGHHII 2 Light("EXILATK1")
			{
				if (ire) tics = 2;
				else tics = random(2, 4);
				A_FaceTarget();
				A_SpawnItemEx("ExileHandFire",8,-32,78,frandom(-1,1),frandom(-1,1),frandom(0,5));
				A_SpawnItemEx("ExileHandFire",8,32,78,frandom(-1,1),frandom(-1,1),frandom(0,5));
			}
			A04_ I 0
			{
				if (pm_attackloop <= 0) {
					SetStateLabel("See");
				}
				else {
					pm_attackloop--;
					A_VileTarget("PM_ImmolatorGeyser");
				}
				
				pm_irecount -= random(2, 6);
			}
			Goto Geysers+1;
		Missile:
			TNT1 A 0 A_Jump(100, "Geysers");
			TNT1 A 0 A_SetSpeed(35);
			A04_ MLK 5 A_FaceTarget();
			A04_ M 0
			{
				for (int i = 0; i < 256; i++)
				{
					A_FaceTarget();
					if (target)
					{
					if (distance3D(target) > 120)
					{
						A_Chase(null, null);
						if (CheckSight(target)) A_SpawnItemEx("PM_ExileFireGlobs",0,0,30,0,0,0,0,128);
					}
					else SetStateLabel("Explode");
					}
				}
				
				pm_irecount -= 5;
			}
			Goto See;
		Melee:
		Explode:
			A04_ M 1
			{
				A_StartSound("GenericChargeHeavy", CHAN_7);
				A_StartSound("Immolator/Attack", CHAN_VOICE);
				A_StartSound("Watcher/ShardCharge",11);
			}
			A04_ MMMLLLKKKMMMLLLKKKMMMLLLKKKMMMLLLKKKMMMLLLKKKMMMLLLKKKMMMLLLKKKJJJ 1 ImmolatorCharge();
			A04_ I 0 ImmolatorExplode();
			A04_ IIHG 10
			{
				for (int i = random(1, 3); i > 0; i--) A_SpawnItemEx("ExileDropFire",random(-150,150),random(-150,150),0,0,0,0,0,SXF_NOCHECKPOSITION);
				A_FaceTarget();
			}
			A04_ GGA 10;
			Goto See;
		Pain:
			A04_ Q 3 { pm_irecount -= random(4, 7); }
			A04_ Q 3 A_Pain();
			Goto See;
		Death:
			TNT1 A 0;
			A04_ Q 5
			{
				A_StopSounds(0,0);
				A_RemoveChildren(1,RMVF_EVERYTHING);
			}
			A04_ R 0 A_StartSound("GenericChargeHeavy", CHAN_7);
			A04_ M 0 A_StartSound("Immolator/Attack", CHAN_VOICE);
			TNT1 A 0 A_StartSound("Watcher/ShardCharge",11);
			A04_ RRRRRRRRSSSSSSTTTTTTTTUUUUUUUUUUUUUUUUUUUUUUUUUUUUUUUUUU 1 ImmolatorCharge();
			A04_ V 0 ImmolatorExplode();
			A04_ VWXYZ 4;
			A04_ Z -1;
			Stop;
	}
}
