Class PM_ViscusMeat : PM_DregMeatSmall
{
	Default
	{
		DamageFunction (10);
		Scale 1.25;
		Translation "0:255=%[0.00,0.00,0.00]:[1.11,0.26,0.26]";
	}
	
	override void Tick()
	{
		PMProjectileBase.Tick();
		if (!level.isFrozen())
		{
			roll += rollp;
			if (launched)
			{
				let Coord = PM_Trailer.Trail(Self,lastpos,"aa0909",(-4,0,0),1.2,0.25,5);
				lastpos=Coord;
			}
			if (!(getAge() % 3)) A_SpawnItemEx("PM_DarkRedBlood1",random(-2,2),random(-2,2),random(-2,2),random(3,6),random(-3,3),frandom(-4,4),flags:SXF_NOCHECKPOSITION);
		}
	}
	
	States
	{
	Spawn:
		R03M AAAAAAAAAAAAAAAAAAAAA 1 { vel *= 0.925; }
		R03M A 10;
		R03M A 0
		{
			A_SpawnParticleEx("",TexMan.CheckForTexture("LENRA0",TexMan.Type_Sprite),STYLE_Add,SPF_ROLL|SPF_RELATIVE|SPF_FULLBRIGHT,random(10,20),100,startalphaf:0.4,sizestep:-1);
			A_SpawnParticleEx("",TexMan.CheckForTexture("DTPRR0",TexMan.Type_Sprite),STYLE_Add,SPF_ROLL|SPF_RELATIVE|SPF_FULLBRIGHT,random(10,20),60,sizestep:-1,startroll:random(0,360));
			A_SeekerMissile(999, 999);
			A_StartSound("Grievance/Launch", 10, pitch: 1.6);
			launched = true;
		}
		R03M A 1 { vel *= 1.13; }
		R03M A 1 A_SeekerMissile(999, 999);
		R03M AAA random(1,2) { vel *= 1.13; }
		Goto LaunchLoop;
	Death:
		TNT1 AAAA 0 A_SpawnItemEx("PM_DarkRedBlood1",random(-2,2),random(-2,2),random(-2,2),random(-5,5),random(-5,5),frandom(-5,5),flags:SXF_NOCHECKPOSITION);
		Stop;
	}
}
