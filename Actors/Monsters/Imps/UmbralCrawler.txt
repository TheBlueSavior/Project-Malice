Class PM_UmbralCrawler : PMUmbralMonsterBase
{
	Default
	{
		PMMonsterBase.CodexLore
		"placeholder text.";
		Health 100;
		Radius 20;
		Height 56;
		Speed 10;
		PainChance 50;
		Mass 100;
		+FLOORCLIP
		+SLIDESONWALLS
		+NOFEAR
		+DONTHARMSPECIES
		+DONTHARMCLASS
		+MISSILEMORE
		+MISSILEEVENMORE
		Monster;
		SeeSound "UmbralCrawler/Idle";
		PainSound "UmbralCrawler/Idle";
		DeathSound "UmbralCrawler/Death";
		ActiveSound "UmbralCrawler/Active";
		Tag "Umbral Crawler";
		Obituary "%o got sniped by a Umbral Crawler.";
		HitObituary "%o got caught off guard by a Umbral Crawler's swipe.";
		Species "Imps";
		Bloodtype "PM_BlackBloodBase";
		TeleFogSourceType "PM_SmallDemonicTeleportFog";
		TeleFogDestType "PM_DemonicTeleportFog";
		MeleeThreshold 120;
		MaxTargetRange 1250;
	}
	
	States
	{
	Spawn:
		I15_ I 0 NoDelay;
	Idle:
		I15_ A 10 PM_Look();
		Loop;
	Look:
		I15_ BBCCDDEE random(2,3)
		{
			A_Recoil(-0.5);
			PM_LookingForPlayer();
		}
		Loop;
	Spotted:
		TNT1 A 0 A_Jump(128,"RoarMissile");
	Roar:
		I15_ H 0 PM_AlertSound();
		Goto See;
	RoarMissile:
		I15_ H 0 PM_AlertSound();
		Goto See;
	See:
		I15_ AABBCCDD random(2,3)
		{
			A_Recoil(-0.5);
			A_FastChase();
		}
		Loop;
	Pain:
		I15_ H 2;
		I15_ H 4 A_Pain();
		Goto See;
	Missile:
	Snipe:
		I15_ AAFF 2 A_FaceTarget();
		I15_ I 0
		{
			A_StartSound("GenericChargeSmall", 12);
			A_StartSound("UmbralCrawler/EyeOpen",CHAN_6);
		}
		I15_ IIII 3 A_FaceTarget();
		I15_ I 0 A_CheckSight("SnipeWait");
		Goto SnipeFire;
	SnipeWait:
		I15_ II 2 A_FaceTarget();
		TNT1 A 0 A_JumpIfTargetInLOS("SnipeFire",360,JLOSF_DEADNOJUMP);
		TNT1 A 0 A_Jump(4,"See");
		Loop;
	SnipeFire:
		I15_ II 2 A_FaceTarget();
		I15_ KKK 3
		{
			A_StartSound("Bacillus/Proj",11,attenuation: 0.8);
			A_StartSound("UmbralCrawler/Fire",10,pitch:frandom(1, 1.5));
			A_FaceTarget();
			A_SpawnProjectile("PM_UmbralCrawlerBall",40);
		}
		I15_ IIIIIIIII 2 A_FaceTarget();
		I15_ A 0 A_StartSound("UmbralCrawler/EyeClose",CHAN_6);
		I15_ FFAA 4 A_FaceTarget();
		Goto See;
	Death:
		I15_ H 5;
		I15_ L 5 A_ScreamAndUnblock();
		I15_ MNO 4 A_SpawnItemEx("PM_BlackInstestinGib",random(-4,4),random(-4,4),random(25,35),random(-3,3),random(-3,3),random(0,5),random(-180,180),SXF_NOCHECKPOSITION);
		I15_ P 0 A_StartSound("Corpse/Fall",CHAN_6);
		I15_ P -1;
		Stop;
	XDeath:
		I15_ Q 5 PM_Gib();
		I15_ R 5 A_XScream;
		I15_ S 5;
		I15_ T 5 A_NoBlocking;
		I15_ U -1;
		Stop;
	Melee:
		I15_ E 10 A_StartSound("UmbralCrawler/Attack", CHAN_VOICE);
		I15_ F 4 A_FaceTarget();
		I15_ G 6
		{
			A_StartSound("Doom4/Other/MeleeSwing",22,CHANF_OVERLAP,pitch:1.75);
			A_Recoil(-10);
			A_SpawnProjectile("PM_FeralLunge",18,flags:CMF_AIMDIRECTION,pitch);
		}
		I15_ F 0 { if (target && Distance3D(target) > 128) SetStateLabel("See"); }
		Loop;
	Raise:
		I15_ PONML 7;
		Goto See;
	}
}
