Class PM_HeavyCavalry : PMSlothMonsterBase
{
	Default
	{
		PMMonsterBase.CodexLore
		"\c[Gray]GUNDAM!!!!!!";
		Health 1000;
		Scale 1.25;
		Tag "Heavy Cavalry";
		BloodType "PM_WarBloodBase";
		Radius 22;
		Height 64;
		Speed 10;
		Mass 1500; //tanky.
		Monster;
		PainChance 25;
		+FLOORCLIP
		+DONTHARMCLASS
		+DONTHARMSPECIES
		+MISSILEMORE
		+MISSILEEVENMORE
		+NORADIUSDMG
		MinMissileChance 50;
		DamageFactor 0.9;
		Species "Nobles";
		SeeSound "HeavyCavalry/Sight";
		PainSound "HeavyCavalry/Pain";
		DeathSound "HeavyCavalry/Death";
		ActiveSound "HeavyCavalry/Idle";
		Obituary "%o faced the torment of war.";
		TeleFogSourceType "PM_SmallDemonicTeleportFog";
		TeleFogDestType "PM_WarTeleportFog";
		PMMonsterBase.RealHeight 80;
	}
	
	override void PostBeginPlay()
	{
		Super.PostBeginPlay();
		tex[0] = TexMan.CheckForTexture("LENRA0",TexMan.Type_Sprite);
		tex[1] = TexMan.CheckForTexture("SPKOA0",TexMan.Type_Sprite);
		tex[2] = TexMan.CheckForTexture("DTPRR0",TexMan.Type_Sprite);
		tex[3] = TexMan.CheckForTexture("GNSMK0",TexMan.Type_Sprite);
	}
	
	TextureID tex[4];
	
	States
	{
	Spawn:
		TERM E 0 NoDelay;
	Idle:
		TERM A 10 PM_Look();
		Loop;
	Look:
		TERM BBBB 2 PM_LookingForPlayer();
		TNT1 A 0
		{
			A_Quake(1, 4, 0, 312);
			A_StartSound("director/servo",9,CHANF_OVERLAP);
			A_StartSound("PMCataphract/Step",9,CHANF_OVERLAP,0.8,pitch:1.2);
		}
		TERM C 15;
		TERM DDDD 2 PM_LookingForPlayer();
		TNT1 A 0
		{
			A_Quake(1, 4, 0, 312);
			A_StartSound("director/servo",9,CHANF_OVERLAP);
			A_StartSound("PMCataphract/Step",9,CHANF_OVERLAP,0.8,pitch:1.2);
		}
		TERM A 15;
		TNT1 A 0 A_JumpIfTargetinLOS("Spotted",250,JLOSF_DEADNOJUMP);
		Loop;
	Spotted:
		TERM A 0 A_Jump(256, "Roar", "NoRoar");
	Roar:
		TERM A 0
		{
			if (!random(0, 1)) A_StartSound("HeavyCavalry/Laugh", CHAN_VOICE);
			else PM_AlertSound();
		}
		TERM AJ 10 A_FaceTarget();
		TERM J 0 A_StartSound("Warforge/SCREAM", 20, volume: 0.65, pitch: 0.85);
		TERM JJJJJJJJKKKKK 2
		{
			Radius_Quake(randompick(1, 1, 2), 2, 0, 10, 0);
			for (int i = -1; i < 2; i += 2)
			{
				A_SpawnParticleEx("", tex[0], STYLE_Add, SPF_RELATIVE|SPF_FULLBRIGHT|SPF_ROLL, 1, random(150,250), xoff: -15, yoff: 20 * i, zoff: 46, startalphaf: 0.45, startroll: random(-180,180));
				A_SpawnParticleEx("", tex[1], STYLE_Add, SPF_RELATIVE|SPF_ROLL, random(55, 80), random(25,45), xoff: -15 + frandom(-5, 5), yoff: 20 * i + frandom(-5, 5), zoff: 46 + frandom(-5, 5), velx: frandom(0, -2), vely: frandom(8, 5) * i, velz: frandom(1.5, 2), accelz: 0.025, startalphaf: frandom(.2, .6), sizestep: frandom(0, -1), startroll: random(-180,180), rollvel: frandom(-1, 1));
				A_SpawnParticleEx("", tex[3], STYLE_Normal, SPF_RELATIVE|SPF_ROLL, random(80, 125), random(35,80), xoff: -15, yoff: 20 * i + frandom(-5, 5), zoff: 35 + frandom(-5, 5), velx: frandom(0, -2), vely: frandom(4, 2) * i, velz: frandom(1, 1.5), accelz: 0.01, startalphaf: frandom(.2, .6), sizestep: frandom(0, 1), startroll: random(-180,180), rollvel: frandom(-4, 4));
			}
		}
		Goto See;
	NoRoar:
		TERM A 0 PM_AlertSound();
		Goto See;
	See:
		TERM BBBB 2 A_Chase();
		TNT1 A 0
		{
			A_Quake(1, 4, 0, 312);
			A_StartSound("director/servo",9,CHANF_OVERLAP);
			A_StartSound("PMCataphract/Step",9,CHANF_OVERLAP,0.8,pitch:1.2);
		}
		TERM C 15;
		TERM DDDD 2 A_Chase();
		TNT1 A 0
		{
			A_Quake(1, 4, 0, 312);
			A_StartSound("director/servo",9,CHANF_OVERLAP);
			A_StartSound("PMCataphract/Step",9,CHANF_OVERLAP,0.8,pitch:1.2);
		}
		TERM A 15;
		Loop;
	}
}
