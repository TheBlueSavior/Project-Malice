Class PM_GildedEffigy : PMGreedMonsterBase
{
	Default
	{
		PMMonsterBase.CodexLore
		"Don't get decieved by those statues, while they look stagnant at first sight, they will wake up when you are nearby it.";
		Health 500;
		Radius 20;
		Height 56;
		Speed 6;
		PainChance 70;
		Mass 800;
		Scale 1.15;
		+FLOORCLIP
		+MISSILEEVENMORE
		+NOGRAVITY
		+FLOAT
		+FLOATBOB
		FloatBobStrength 0.2;
		MeleeDamage (35);
		Monster;
		SeeSound "GildedEffigy/Sight";
		PainSound "GildedEffigy/Pain";
		DeathSound "GildedEffigy/Death";
		ActiveSound "GildedEffigy/Active";
		Tag "Gilded Effigy";
		Obituary "%o got turned into golden statue by a Gilded Effigy.";
		HitObituary "%o is now converted to molten gold.";
		Species "Cacodemon";
		//Bloodtype "PM_GreedBloodBase";
		BloodColor "Yellow";
		TeleFogSourceType "PM_SmallDemonicTeleportFog";
		TeleFogDestType "PM_DemonicTeleportFog";
		StencilColor "Yellow";
		MeleeRange 170;
	}
    
	bool active;
	
	void setActive(bool choice = true)
	{
		A_StartSound((choice) ? "GildedEffigy/Activate" : "GildedEffigy/Deactivate");
		
	}
	
	States
	{
	Spawn:
		C04_ A 0 nodelay;
	Idle:
		C04_ L 10 PM_Look();
		Loop;
	Look:
		C04_ L 5
		{
			if (target && distance3D(target) < 400)
				SetStateLabel("Spotted");
		}
		Loop;
	Spotted:
		TNT1 A 0 A_Jump(256, "GoActive", "GoInactive");
	GoActive:
		C04_ H 0
		{
			PM_AlertSound();
			setActive();
		}
		Goto ActiveSee;
	GoInactive:
		C04_ H 0;
		Goto See;
	See:
		C04_ L 5
		{
			if (target && !checkSight(target))
			{
				tics = random(1,3);
				for (int i = 0; i < random(1, 5); i++)
				{
					if (!random(0, 1)) A_Wander();
					else A_Chase();
				}
				A_Recoil(-0.4);
			}
			else
			{
				A_Stop();
				A_Chase(flags:CHF_DONTMOVE|CHF_DONTTURN);
			}
		}
		Loop;
	ActiveSee:
		C04_ A 2
		{
			A_Recoil(-0.1);
			A_Chase();
		}
		Loop;
	Pain:
		C04_ E 3;
		C04_ E 3 A_Pain();
		Goto See;
	Death:
		C04_ F 0;
		C04_ F 8
		{
			bFLOATBOB = false;
		}
		C04_ G 8 A_ScreamAndUnblock();
		C04_ HIJ 7;
		C04_ K -1;
		Stop;
	}
}
