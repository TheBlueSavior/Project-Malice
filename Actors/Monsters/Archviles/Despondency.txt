Class PM_Despondency : PMSlothMonsterBase
{
	Default
	{
		PMMonsterBase.CodexLore
		"\c[cyan]The depression of spirits from loss of courage or hope. The universe is infinite, yet there is no hope. I can't. We all can't. Let us all rot in this abyssmal place.\c-
		
		They are infusing despair to mortals, even though that might not fix their own despair nor lethargy. Although it might be a little lethargic, it is more quicker in contrast to other Sloth demons and focuses on cosmic and celestial energy infused with hellish energy.
		
		Put it out of its misery before it puts it onto you as well. It can summon a huge black hole.";
		Tag "Despondency";
		Scale 1.2;
		Health 900;
		Radius 20;
		Height 56;
		Mass 500;
		Speed 8;
		PainChance 40;
		//MaxTargetRange 896;
		Species "Archvile";
		DamageFactor "Vile", 0;
		PainChance "Vile", 0;
		Gibhealth 40;
		Obituary "%o fully lost hope.";
		SeeSound "PMDespondency/Sight";
		PainSound "PMDespondency/Pain";
		DeathSound "PMDespondency/Death";
		ActiveSound "PMDespondency/Active";
		Bloodtype "PM_SlothBloodBase";
		FloatSpeed 2;
		+NOTARGET
		+DONTHARMSPECIES
		+DONTHARMCLASS
		+FLOAT
		+NOGRAVITY
		+NOINFIGHTSPECIES
		+NOINFIGHTING

		+MISSILEEVENMORE
		TeleFogSourceType "PM_SmallDemonicTeleportFog";

		PMMonsterBase.RealHeight 70;
		StencilColor "Cyan";
		MeleeRange 350;
		DamageFactor "SlothVile", 0.00;
	}
	
	textureID tex[7];
	
	override void PostBeginPlay()
	{
		Super.PostBeginPlay();
		tex[0] = TexMan.CheckForTexture("CBALC0",TexMan.Type_Sprite);
		tex[1] = TexMan.CheckForTexture("GNSML0",TexMan.Type_Sprite);
		tex[2] = TexMan.CheckForTexture("SPKOS0",TexMan.Type_Sprite);
		tex[3] = TexMan.CheckForTexture("DUSTC0",TexMan.Type_Sprite);
		tex[4] = TexMan.CheckForTexture("GNSMK0",TexMan.Type_Sprite);
		tex[5] = TexMan.CheckForTexture("FIR4R0",TexMan.Type_Sprite);
		tex[6] = TexMan.CheckForTexture("STR_A0",TexMan.Type_Sprite);
	}
	
	States
	{
	Spawn:
		A05_ B 0;
		Goto Idle;
	Idle:
		A05_ ABCD 10 PM_Look();
		Loop;
	Look:
		A05_ AAABBB 5 PM_LookingForPlayer();
		TNT1 A 0 A_JumpifTargetInLOS("Spotted",180,JLOSF_DEADNOJUMP);
		A05_ CCCDDD 5 PM_LookingForPlayer();
		TNT1 A 0 A_CheckSight("See");
		Goto Spotted;
	Spotted:
		TNT1 A 0 PM_AlertSound();
	See:
		A05_ AAABBBCCCDDD random(3,4)
		{
			A_SpawnItemEx("PM_GenericPhaseGhost",flags:SXF_TRANSFERSPRITEFRAME|SXF_TRANSFERSCALE|SXF_TRANSFERSTENCILCOL);
			A_VileChase();
		}
		Loop; 
	Raise:
		A05_ ALKKKKL 4;
		Goto See;
	Pain:
		A05_ P 4;
		A05_ P 4 A_Pain();
		Goto See;
	Melee:
	Mininova:
		A05_ A 0 A_StartSound("PMDespondency/BHCharge", 30);
		A05_ ONONONONONONONONLK 2
		{
			A_FaceTarget();
			
			BlockThingsIterator ti = BlockThingsIterator.Create(self, 600);
			actor a;
			
			while(ti.Next())
			{
				a = ti.thing;
				double rad = abs((600 - Distance3D(a)) / 2);
				if (a && a.Health > 0 && a != self && a.bSHOOTABLE && !a.bINVULNERABLE && !a.bDORMANT && !a.bNODAMAGE && CheckSight(a) && distance3D(a) < 600 && !a.bDONTTHRUST)
					a.VelFromAngle(-(rad / 25), self.AngleTo(a));
			}
			
			
			A_QuakeEx(1,1,1,10,0,600,"",QF_3D|QF_SCALEDOWN);
			A_SpawnParticleEx("7affe0", TexMan.CheckForTexture("SPKWE0"), style: STYLE_Add, flags: SPF_RELATIVE|SPF_FULLBRIGHT|SPF_ROLL, lifetime: random(1,5), size: frandom(250,500), xoff: frandom(-50,50), yoff: frandom(-50,50), velz: frandom(0, 10), startalphaf: frandom(.4,1), fadestepf: 0.4, sizestep: -20, startroll: random(-180,180), rollvel: frandom(-4, 4));
			A_SpawnParticleEx("", TexMan.CheckForTexture("SPKBA0"), style: STYLE_Add, flags: SPF_RELATIVE|SPF_FULLBRIGHT|SPF_ROLL, lifetime: random(10,15), size: frandom(350,500), xoff: frandom(-50,50), yoff: frandom(-50,50), velz: frandom(0, 5), startalphaf: frandom(.4,1), fadestepf: 0.4, sizestep: -20, startroll: random(-180,180), rollvel: frandom(-4, 4));
			A_SpawnParticleEx("00ff78", TexMan.CheckForTexture("SPKBA0"), style: STYLE_ADD, flags: SPF_RELATIVE|SPF_ROLL|SPF_FULLBRIGHT, lifetime: random(50,100), size: frandom(0.5,15), xoff: random(-90,90), yoff: random(-90,90), zoff: random(10,50), accelx: -1, startalphaf: 1.0, fadestepf: 0.1, sizestep: -0.4, startroll: random(-180,180));
		}
		A05_ K 0
		{
			A_StartSound("PMDespondency/Shockwave1", 31);
			A_StartSound("PMDespondency/Shockwave2", 32);
			A_StartSound("DEmi/CannonFire", 33, pitch: 1.35);
			Radius_Quake (8, 40, 0, 10, 0);
		}
		A05_ KKKKKKKKKKKKKKKKKKK 1
		{
			A_FaceTarget();
			A_Explode(5, 200, XF_NOTMISSILE);
			A_SpawnParticleEx("7affe0", TexMan.CheckForTexture("SPKWE0"), style: STYLE_Add, flags: SPF_RELATIVE|SPF_FULLBRIGHT|SPF_ROLL, lifetime: random(1,5), size: frandom(250,500), xoff: frandom(-50,50), yoff: frandom(-50,50), velz: frandom(0, 10), startalphaf: frandom(.4,1), fadestepf: 0.4, sizestep: -20, startroll: random(-180,180), rollvel: frandom(-4, 4));

			A_SpawnParticleEx("00ff78", TexMan.CheckForTexture("SPKBA0"), style: STYLE_ADD, flags: SPF_RELATIVE|SPF_ROLL|SPF_FULLBRIGHT, lifetime: random(50,100), size: frandom(0.5,15), xoff: random(-90,90), yoff: random(-90,90), zoff: random(10,50), accelx: -1, startalphaf: 1.0, fadestepf: 0.1, sizestep: -0.4, startroll: random(-180,180));
			A_SpawnParticleEx("", TexMan.CheckForTexture("CBALC0"), style: STYLE_Add, flags: SPF_RELATIVE|SPF_FULLBRIGHT|SPF_ROLL, lifetime: 1, size: frandom(50,1000), zoff: 50, startalphaf: frandom(.4,1), fadestepf: 0.4, startroll: random(-180,180), rollvel: frandom(-4, 4));
		}
		A05_ K 30
		{
			A_StopSound(CHAN_VOICE);
			A_StartSound("Famine/Explode",CHAN_5,volume:0.85,pitch:frandom(0.8,1));
			A_StartSound("PMDespondency/Amb2",20,pitch:1.6);
			A_StartSound("PMDespondency/Amb1",21,pitch:1.6);
			A_StartSound("PMDespondency/Explode2",CHAN_6,attenuation:0.85,pitch:0.7);
			A_StartSound("PMDespondency/Explode",CHAN_7,pitch:1.3);
			A_StartSound("DEmi/CannonFire", 33, attenuation:0.5, pitch: 1.35);
			A_Recoil(10);
			vel.z += 10;
			A_Explode((40),200,flags:XF_NOTMISSILE,damagetype:"SlothVile");
			A_QuakeEx(5,5,5,10,0,400,0,0,0,0,0,150);
			
			for (int i = 6; i > 0; i--)
			{
				let fx1 = Spawn("PM_GenericExplosionFlames", Vec3Offset(0,0,0));
				if (fx1)
				{
					fx1.A_SetRenderStyle(1, STYLE_ADDSHADED);
					fx1.SetShade("34abeb");
					fx1.vel = (frandom(-19,19),frandom(-19,19),frandom(-9,9));
				}
			}
			
			BlockThingsIterator ti = BlockThingsIterator.Create(self, 450);
			actor a;
			
			while(ti.Next())
			{
				a = ti.thing;
				double rad = abs((450 - Distance3D(a)) / 2);
				if (a && a.Health > 0 && a != self && a.bSHOOTABLE && !a.bINVULNERABLE && !a.bDORMANT && !a.bNODAMAGE && CheckSight(a) && distance3D(a) < 450 && !a.bDONTTHRUST)
				{
					a.Vel3DFromAngle((rad / 2) / max(1, a.Mass / 2), self.AngleTo(a), self.PitchTo(a));
					if (a.GetMaxHealth() < 2500)
						a.vel.z += (rad / 10);
					a.vel.z = clamp(a.vel.z, -25, 25);
				}
			}
			
			A_StartSound("PMBelphegor/TwilightExplode", 50, attenuation:0.5);
			for(int i=random(6, 16); i>0; i--) A_SpawnParticleEx("40acff", tex[4], style: STYLE_Add, flags: SPF_ROLL|SPF_RELATIVE|SPF_FULLBRIGHT, lifetime: random(150,600), size: frandom(100, 200), zoff: 50, velx: frandom(-1, 1), vely: frandom(-1, 1), velz: frandom(-1, 1), startalphaf: frandom(.15,.3), fadestepf: -1, sizestep: 5, startroll: random(-180,180), rollvel: frandom(-1, 1));

			for(int i=random(15,40);i>0;i--) A_SpawnParticleEx("727272",tex[3],style: STYLE_Normal,flags: SPF_RELATIVE|SPF_ROLL,lifetime: random(40,70),size: frandom(2,35),xoff: frandom(-20, 20), yoff: frandom(-20, 20), zoff: frandom(-20, 20) + 50,velx: frandom(-8,8),vely: frandom(-8,8),velz: frandom(-8,12),accelz: frandom(-0.1,-0.5),startalphaf: 1.0,sizestep: -0.2,startroll: random(-180,180),rollvel: random(-4,4));
			for(int i=random(5, 10); i>0; i--) A_SpawnParticleEx("0c2036", tex[1], style: STYLE_Add, flags: SPF_ROLL|SPF_RELATIVE|SPF_FULLBRIGHT, lifetime: random(100,200), size: frandom(100, 200), xoff: frandom(-20, 20), yoff: frandom(-20, 20), zoff: frandom(-20, 20) + 50, velx: frandom(-1, 1), vely: frandom(-1, 1), velz: frandom(-1, 1), startalphaf: frandom(0.2,1), fadestepf: -1, sizestep: 2, startroll: random(-180,180), rollvel: frandom(-1, 1));
			
			A_Explode((40),200,XF_NOTMISSILE);
			A_QuakeEx(5,5,5,10,0,400,0,0,0,0,0,150);
			vel *= 0.00;
		}
		Goto See;
	Death:
		A05_ P 5;
		A05_ Q 6
		{
			bNOGRAVITY = false;
			bFLOAT = false;
			A_ScreamAndUnblock();
			A_SpawnItemEx("PM_SlothMeatChunkLauncher",random(-2,2),random(-2,2),random(10,40),0,0,0,0,SXF_NOCHECKPOSITION,60);
		}
		A05_ RST 6 A_SpawnItemEx("PM_SlothMeatChunkLauncher",random(-2,2),random(-2,2),random(10,40),0,0,0,0,SXF_NOCHECKPOSITION,60);
		A05_ UVW 7;
		A05_ X 5 A_StartSound("Corpse/Fall",CHAN_BODY);
		A05_ X -1;
		Stop;
	}
}
