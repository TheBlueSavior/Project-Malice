	override void Tick()
	{
		Super.Tick();
		if (health <= 0)
		{
			A_StopSound(30);
			A_StopSound(31);
			A_StopSound(32);
			return;
		}
		if (target && distance3D(target) <= 1800) A_StartSound("Warmonger/Engine", 30, CHANF_LOOPING);
		else A_StopSound(30);

		if (--exhaustTimer <= 0)
		{
			exhaustTimer = random(2, 4);
			let cs = CurState;
			if (cs && !InStateSequence(cs, FindState("Spawn")) && !InStateSequence(cs, FindState("Idle")) && !InStateSequence(cs, FindState("Look")))
			{
				A_SpawnParticleEx("555555", TexMan.CheckForTexture("GNSML0"),
					style: STYLE_Normal, flags: SPF_RELATIVE | SPF_ROLL,
					lifetime: random(50, 80), size: frandom(8, 16),
					xoff: -20, yoff: 0, zoff: 65,
					velx: frandom(-0.8, 0.3), vely: frandom(-0.5, 0.5), velz: frandom(2.5, 4.0),
					accelz: -0.03,
					startalphaf: frandom(0.4, 0.65), fadestepf: -1,
					sizestep: frandom(0.5, 1.0), startroll: random(-180, 180), rollvel: frandom(-1.2, 1.2));
			}
		}
	}

	void WarLockLaser(Vector3 aimPos)
	{
		double sa = sin(angle);
		double ca = cos(angle);
		Vector2 relOff = (24 * ca + 18 * sa, 24 * sa - 18 * ca);
		PM_SpawnBolt(self, aimPos, "ff0000", TexMan.CheckForTexture("RLSSA0"), maxDiff: 0, scale: 0.25, offset: (relOff.X, relOff.Y, 44), baseTime: 2, fadeTime: 0);
	}

	void WarRailLaser()
	{
		FLineTraceData lt;
		LineTrace(angle, 8192, pitch, 0, 44, 20, 18, lt);
		double sa = sin(angle);
		double ca = cos(angle);
		Vector2 relOff = (20 * ca + 18 * sa, 20 * sa - 18 * ca);
		PM_SpawnBolt(self, lt.HitLocation, "ff0000", TexMan.CheckForTexture("RLSSA0"), maxDiff: 0, scale: 0.25, offset: (relOff.X, relOff.Y, 44), baseTime: 3, fadeTime: 0);
	}

	void RailTrack(double maxstep)
	{
		if (!target) return;
		double ca = cos(angle);
		double sa = sin(angle);
		Vector3 muzzle = level.Vec3Offset(pos, (20 * ca + 18 * sa, 20 * sa - 18 * ca, 44));
		Vector3 diff = level.Vec3Diff(muzzle, target.Vec3Offset(0, 0, target.Height * 0.55));
		double desAng = atan2(diff.y, diff.x);
		double desPit = -atan2(diff.z, diff.xy.Length());
		angle += clamp(deltaangle(angle, desAng), -maxstep, maxstep);
		pitch += clamp(deltaangle(pitch, desPit), -maxstep, maxstep);
	}

	void RailgunBeam()
	{
		double ca = cos(angle);
		double sa = sin(angle);
		Vector3 muzzle = level.Vec3Offset(pos, (20 * ca + 18 * sa, 20 * sa - 18 * ca, 44));
		FLineTraceData lt;
		LineTrace(angle, 8192, pitch, TRF_SOLIDACTORS, 44, 20, 18, lt);
		Vector3 d = level.Vec3Diff(muzzle, lt.HitLocation);
		int steps = int(d.Length() / 28);
		if (steps < 1) steps = 1;
		for (int i = 0; i <= steps; i++)
		{
			Vector3 p = level.Vec3Offset(muzzle, d * (double(i) / double(steps)));
			let seg = Spawn("PM_WarmongerRailgunBeamFX", p);
			if (seg) { seg.angle = angle; seg.pitch = pitch; }
		}
		Spawn("PM_WarmongerRailgunImpact", lt.HitLocation);
		if (lt.HitActor && lt.HitActor.bShootable && lt.HitActor != self)
			lt.HitActor.DamageMobj(self, self, 200, "Railgun");
	}

	void RadTick()
	{
		for (int i = 0; i < MAXPLAYERS; i++)
		{
			if (!playeringame[i] || !players[i].mo) continue;
			let pmo = players[i].mo;
			if (pmo.health <= 0 || Distance3D(pmo) > 200 || !CheckSight(pmo)) continue;
			pmo.DamageMobj(self, self, 6, "Radiation", DMG_THRUSTLESS);
			if (pmo.health > 0) pmo.GiveInventory("PM_WarmongerIrradiated", 1);
		}
	}

	void TBPaintFirst()
	{
		tbSpot = null;
		if (!target) return;
		Vector2 dir2 = level.Vec2Diff(pos.xy, target.pos.xy);
		double d2 = dir2.Length();
		Vector2 u2 = d2 > 1 ? dir2 / d2 : (cos(angle), sin(angle));
		double along = random(0, 3) ? -frandom(16, 100) : frandom(48, 200);
		Vector2 mark2 = target.pos.xy + u2 * along + (frandom(-40, 40), frandom(-40, 40));
		Vector3 muzzle = Vec3Offset(0, 0, 46);
		Vector3 dv = level.Vec3Diff(muzzle, (mark2.x, mark2.y, target.floorz + 2));
		FLineTraceData lt;
		LineTrace(atan2(dv.y, dv.x), 1400, -atan2(dv.z, dv.xy.Length()), TRF_THRUACTORS, 46, 0, 0, lt);
		let s = Spawn("PM_ThunderSpot", lt.HitLocation, ALLOW_REPLACE);
		if (s)
		{
			s.target = self;
			s.SetZ(clamp(s.pos.z, s.floorz, s.ceilingz - 8));
			tbSpot = s;
		}
	}

Class PM_WarmongerRadOverlay : EventHandler
{
	override void RenderOverlay(RenderEvent e)
	{
		let pmo = players[consoleplayer].mo;
		if (!pmo) return;
		let pw = Powerup(pmo.FindInventory("PM_WarmongerIrradiated"));
		if (!pw) return;

		double strength = clamp(pw.EffectTics / 280.0, 0.12, 1.0);
		double sw = double(Screen.GetWidth());
		double sh = double(Screen.GetHeight());
		double px = max(1.0, sh / 400.0);
		int count = int(18 + 72 * strength);
		for (int i = 0; i < count; i++)
		{
			int h = (gametic * 374761393 + i * 668265263) & 0x7fffffff;
			h = ((h ^ (h >> 13)) * 1274126177) & 0x7fffffff;
			h ^= h >> 16;
			double x = (h % 8191) / 8191.0 * sw;
			double y = ((h >> 13) % 8191) / 8191.0 * sh;
			double a = (0.2 + ((h >> 26) % 16) / 24.0) * strength;
			int s = int(px * (1 + (h >> 21) % 3));
			int tall = (h % 7 == 0) ? s * 3 : s;
			Screen.Dim("ffffff", a, int(x), int(y), s, tall);
		}
	}
}

		{
			Vector3 seg = level.Vec3Diff(nodes[n-1], nodes[n]);
			int steps = max(1, int(seg.Length() / 3));

			{
				let a = it.thing;
				if (a && a != target && a.bSHOOTABLE && a.health > 0 && random(0, 1)
					&& abs(a.pos.z - pos.z) < 96 && Distance2D(a) < 180 && CheckSight(a))
				{
					ArcWalk(a);
					a.A_StartSound("Sybarite/NetShock", CHAN_BODY, CHANF_OVERLAP, 0.6, pitch: frandom(0.8, 1.1));
					a.DamageMobj(self, target, random(4, 7), "Electric", DMG_THRUSTLESS);
				}
			}
		}
		TNT1 A 0 { if (GetAge() >= 6 * 35) SetStateLabel("Fizzle"); }
		Loop;
	Fizzle:
		TNT1 AAAAA 6
		{
			if (!random(0, 1))
				A_SpawnParticleEx("", tex[1], style: STYLE_Add, flags: SPF_FULLBRIGHT|SPF_ROLL,
					lifetime: random(1, 3), size: random(10, 26),
					xoff: frandom(-30, 30), yoff: frandom(-30, 30), zoff: frandom(0, 12),
					startalphaf: frandom(0.2, 0.5), startroll: random(-180, 180));
		}
		Stop;
	}
