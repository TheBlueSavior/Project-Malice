Class PM_Cultist : PMPrideMonsterBase
{ 
	Default
	{
	 PMMonsterBase.CodexLore
	 "Cultists were the most devout of the doomsday sects that worshipped sin long before Hell ever breached the surface. When the circle of Pride finally answered their prayers, it rewarded their faith with a splinter of its arcane might -- enough to raise them above the rabble, and far too little for them to ever truly matter.

	  Certain of their own ascension and drunk on borrowed power, they fling conjured fire and slip between spaces as though the world bends to their will. It does not. Pride simply lets them believe so.

	  Looses rapid bursts of fireballs, and will blink a short distance to reposition when pressed.";
     Health 80;
     Radius 20;
     Height 56;
     Speed 10;
     Mass 100;
     PainChance 100;
     Damage 1;
     Monster;
     +FLOORCLIP
	 +SLIDESONWALLS
	 +HARMFRIENDS
	 +DOHARMSPECIES
	 +MISSILEMORE
	 Species "Zombie";
	 Tag "Cultist";
     Obituary "%o cannot fathom the power of the all-mighty Cultist!";
     SeeSound "Cultist/Sight";
	 PainSound "Cultist/Pain";
	 DeathSound "Cultist/Death";
	 ActiveSound "Cultist/Active";
	 Bloodtype "PM_DarkRedBloodBase";
     DropItem "Clip", 150;
	 DropItem "Shell";
     Decal "Bulletchip";
	 TeleFogSourceType "PM_PossessedTeleportFog";
	 TeleFogDestType "PM_PossessedTeleportFog";
	}
    States
    {
    Spawn:
        S10_ A 0 NoDelay { user_numbullets = random(20,41); }
    Idle:
		S10_ A 10 PM_Look();
        Loop;
	Look:
		TNT1 A 0 A_StartSound("ZombieSoldier/Step",CHAN_6,pitch:frandom(1.1,1.3));
        S10_ AA 4 PM_LookingForPlayer();
		S10_ BB 4 PM_LookingForPlayer();
		TNT1 A 0 A_JumpIfTargetinLOS("Spotted",150,JLOSF_DEADNOJUMP);
		TNT1 A 0 A_StartSound("ZombieSoldier/Step",CHAN_6,pitch:frandom(1.1,1.3));
		S10_ CC 4 PM_LookingForPlayer();
		S10_ DD 4 PM_LookingForPlayer();
        Loop;
	Spotted:
	    TNT1 A 0 A_Jump(256,"Roar","SpotMissile","See");
		Goto See;
	Roar:
	    S10_ E 10 PM_AlertSound();
		Goto See;
	SpotMissile:
        TNT1 A 0 PM_AlertSound();
	    S10_ EEE 3 A_FaceTarget();
	    Goto Missile;
    See:
		TNT1 A 0 { bFRIGHTENED = false; }
		TNT1 A 0 A_SetSpeed(8);
		TNT1 A 0 A_StartSound("ZombieSoldier/Step",CHAN_6,pitch:frandom(1.1,1.3));
        S10_ AA random(1,2) A_Chase();
		TNT1 AA 0 A_Recoil(-0.3);
		S10_ BB random(1,2) A_Chase();
		TNT1 AA 0 A_Recoil(-0.3);
		TNT1 A 0 A_StartSound("ZombieSoldier/Step",CHAN_6,pitch:frandom(1.1,1.3));
		S10_ CC random(1,2) A_Chase();
		TNT1 AA 0 A_Recoil(-0.3);
		S10_ DD random(1,2) A_Chase();
		TNT1 AA 0 A_Recoil(-0.3);
		TNT1 A 0 A_CheckSight("See");
		TNT1 A 0 A_Jump(128,"SeeWander");
        Loop;
	SeeWander:
		TNT1 A 0 { bFRIGHTENED = false; }
		TNT1 A 0 A_SetSpeed(8);
		TNT1 A 0 A_StartSound("ZombieSoldier/Step",CHAN_6,pitch:frandom(1.1,1.3));
        S10_ AA random(1,3)
		{
			A_Chase("Missile","Missile",CHF_DONTMOVE|CHF_DONTTURN);
			A_Wander();
		}
		TNT1 AA 0 A_Recoil(-0.3);
		S10_ BB random(1,3)
		{
			A_Chase("Missile","Missile",CHF_DONTMOVE|CHF_DONTTURN);
			A_Wander();
		}
		TNT1 AA 0 A_Recoil(-0.3);
		TNT1 A 0 A_StartSound("ZombieSoldier/Step",CHAN_6,pitch:frandom(1.1,1.3));
		S10_ CC random(1,3)
		{
			A_Chase("Missile","Missile",CHF_DONTMOVE|CHF_DONTTURN);
			A_Wander();
		}
		TNT1 AA 0 A_Recoil(-0.3);
		S10_ DD random(1,3)
		{
			A_Chase("Missile","Missile",CHF_DONTMOVE|CHF_DONTTURN);
			A_Wander();
		}
		TNT1 AA 0 A_Recoil(-0.3);
		TNT1 A 0 A_JumpIfTargetinLOS("SeeWander",150,JLOSF_DEADNOJUMP);
        Goto See;
    Missile:
        TNT1 A 0;
		TNT1 A 0 A_Jump(40, "Teleportation");
	RealMissile:
		S10_ E 18 
		{
		   if(random(1,2) == 1)
		    A_StartSound("Cultist/Attack",12);
		   A_StartSound("GenericChargeSmall",20);
		   pm_attackloop = random(2,5);
		   A_FaceTarget();
		   A_SetAngle(angle+random(-10,10));
		   A_SetPitch(pitch+random(-18,18));
		}
	RMissileLoop:
        S10_ F 5 BRIGHT Light("ShadowBall2") 
		{
		  if(!pm_attackloop)
		   { SetStateLabel("RMissileEnd"); }
		  for (int i = 0; i < 5; i += 1)
			A_SpawnProjectile("PM_BlashemerBHBall", 32, 0, frandom(-15, 15), CMF_OFFSETPITCH|CMF_AIMDIRECTION, frandom(-15, 15));
          user_numbullets--;
		  pm_attackloop--;
		}
		S10_ EEE 5 A_FaceTarget(0.05,3);
		Loop;
	RMissileEnd:
		S10_ E 10 A_FaceTarget(10,10);
        Goto See;
	Teleportation:
		S10_ AE 10 A_FaceTarget();
		TNT1 A 0
		{
			A_StartSound("HFuryRushAdd", CHAN_6, pitch: 1.5);
			
			bSHOOTABLE = false;
			
			A_SpawnParticleEx
			(
				"",
				TexMan.CheckForTexture("DTPRR0"),
				style: STYLE_Add,
				flags: SPF_RELATIVE|SPF_FULLBRIGHT|SPF_ROLL,
				lifetime: 30,
				size: 450,
				zoff: 40,
				startalphaf: 0.95,
				fadestepf: -1,
				sizestep: -50,
				startroll: random(-180,180),
				rollvel: frandom(-10,10),
				rollacc: frandom(-1,1)
			);
				
			for (int i = random(7, 15); i > 0; i--)
			{		
				A_SpawnParticleEx
				(   
					 "",
					 TexMan.CheckForTexture("SPKRA0"),
					 style: STYLE_Add,
					 flags: SPF_ROLL|SPF_RELATIVE|SPF_FULLBRIGHT,
					 lifetime: 50,
					 size: frandom(25,35),
					 xoff: frandom(-3.5,3.5),
					 yoff: frandom(-3.5,3.5),
					 zoff: 40,
					 velx: frandom(-1,1),
					 vely: frandom(-1,1),
					 velz: frandom(-1,1),
					 startalphaf: 1,
					 fadestepf: -1,
					 sizestep: -2
				);
			}
		}
		TNT1 A 0 ThrustThingZ(0,80,1,0);
		TNT1 AAAAAAAAAAAAAAAAAAAAAAA 1
		{
			for (int i = 4; i > 0; i--) A_Wander();
		}
		TNT1 A 0
		{
			A_StartSound("HFuryRushAdd", CHAN_6, pitch: 1.5);
			
			bSHOOTABLE = true;
			
			A_SpawnItemEx("PM_PossessedTeleportFog",0,0,0,0,0,0,0,SXF_NOCHECKPOSITION);
				
			for (int i = random(7, 15); i > 0; i--)
			{		
				A_SpawnParticleEx
				(   
					 "",
					 TexMan.CheckForTexture("SPKRA0"),
					 style: STYLE_Add,
					 flags: SPF_ROLL|SPF_RELATIVE|SPF_FULLBRIGHT,
					 lifetime: 50,
					 size: frandom(25,35),
					 xoff: frandom(-3.5,3.5),
					 yoff: frandom(-3.5,3.5),
					 zoff: 40,
					 velx: frandom(-1,1),
					 vely: frandom(-1,1),
					 velz: frandom(-1,1),
					 startalphaf: 1,
					 fadestepf: -1,
					 sizestep: -2
				);
			}
		}
		TNT1 A 0 A_Jump(155,"See");
		Goto Missile;
    Pain:
        S10_ G 5;
        S10_ G 3 A_Pain;
        Goto See;
    Death:
        S10_ H 5 A_NoBlocking;
		TNT1 A 0 A_StopSound(555);
        S10_ I 5 PM_Scream();
        S10_ J 5;
        S10_ KL 5;
		TNT1 A 0 A_StartSound("Corpse/Fall",CHAN_6);
		S10_ M 90;
		TNT1 A 0 { if(random(1,3) == 1) A_SpawnItemEx("PM_PossessingDemon",0,0,5,0,0,0,random(0,360),SXF_NOCHECKPOSITION); }
        S10_ M -1;
        Stop;
    XDeath:
		TNT1 A 0 A_StopSound(555);
        S10_ N 5 PM_Gib();
        S10_ O 5 A_XScream;
		S10_ P 5;
        S10_ Q 5 A_NoBlocking;
        S10_ RSTUV 5;
        S10_ W -1;
        Stop;
    Raise:
        S10_ MLKJIH 5;
        Goto Look;
    }
}
