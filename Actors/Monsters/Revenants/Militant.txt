Class PM_Militant : PMWarMonsterBase
{
  int user_leapstrength;
  Default
  {
  PMMonsterBase.CodexLore
  "placeholder.";
  Health 600;
  Radius 20;
  Height 56;
  Mass 500;
  Speed 10;
  PainChance 80;
  MeleeRange 120;
  Monster;
  +FLOORCLIP
  +DONTHARMCLASS
  +DONTHARMSPECIES
  +DONTHARMCLASS
  Species "Revenant";
  SeeSound "Militant/Sight";
  PainSound "Militant/Pain";
  DeathSound "Militant/Death";
  ActiveSound "Militant/Active";
  MeleeSound "Husk/Melee";
  HitObituary "%o has been dismissed by the Militant.";
  Obituary "%o was put to a halt by the Militant.";
  MeleeThreshold 60;;
  Tag "Militant";
  //PoisonDamage 128;
  Bloodtype "PM_WarBloodBase";
  TeleFogSourceType "PM_SmallDemonicTeleportFog";
  TeleFogDestType "PM_WarTeleportFog";
  }
  
  States
  {
  Spawn:
	R10_ G 0 NoDelay;
  Idle:
	R10_ AAAAAAAAAA 10 PM_Look();
	R10_ A 0
	{
	if (!random(0, 30))
			{
				A_StartSound(activesound,554,volume:frandom(0.1,0.5),pitch:frandom(0.60,0.78));
			}
	}
	R10_ A 0 A_Jump(45,"Explore");
    Loop;
  Explore:
	TNT1 A 0
	{
		if (!random(0, 12))
			A_StartSound(activesound,554,volume:frandom(0.1,0.5),pitch:frandom(0.60,0.78));
	}
	TNT1 A 0 A_SetSpeed(3);
	TNT1 A 0 A_StartSound("skeleton/step",4,volume:0.4);
	R10_ AAAABBBB 3
	{
	PM_Look();
	A_Wander();
	}
	TNT1 A 0 A_StartSound("skeleton/step",4,volume:0.4);
	R10_ CCCCDDDD 3
	{
	A_Wander();
	PM_Look();
	}
	R10_ A 0 A_Jump(40,"Idle");
    Loop;
  Look:
	TNT1 A 0 A_SetSpeed(10);
	TNT1 A 0 A_JumpIfTargetInLOS("Spotted",180,JLOSF_DEADNOJUMP);
    R10_ AAABBB 3 PM_LookingForPlayer();
	NULL A 0 A_StartSound("skeleton/step",5);
	TNT1 A 0 A_JumpIfTargetInLOS("Spotted",180,JLOSF_DEADNOJUMP);
	R10_ CCCDDD 3 PM_LookingForPlayer();
	NULL A 0 A_StartSound("skeleton/step",5);
    Loop;
  Spotted:
    TNT1 A 0 A_Jump(60,"SpotMissile");
    Goto Roar;
  Roar:
    TNT1 A 0 PM_AlertSound();
    Goto See;
  SpotMissile:
    TNT1 A 0 PM_AlertSound();
    R10_ A 10 A_FaceTarget;
    Goto Missile;
  See:
	TNT1 A 0 A_SetSpeed(10);
	TNT1 A 0 A_Jump(80,"Sprint");
    R10_ AAABBB random(2,3) A_Chase();
	NULL A 0 A_StartSound("skeleton/step",5);
	R10_ CCCDDD random(3,4) A_Chase();
	NULL A 0 A_StartSound("skeleton/step",5);
    Loop;
  Sprint:
	TNT1 A 0 A_SetSpeed(25);
    R10_ AABB 2
	{
	A_Chase();
    A_SpawnItemEx("PM_FlameSmoke",random(-10,10),random(-16,16),random(25,65),0,0,frandom(1,3));
	A_SpawnItemEx("PM_OrangeEmbersFloat",0,0,random(25,65),FRandom(2,-2),FRandom(2,-2),FRandom(2,-2),0,128,220);
	}
	NULL A 0 A_StartSound("skeleton/step",5);
	R10_ CCDD 2
	{
	A_Chase();
	A_SpawnItemEx("PM_FlameSmoke",random(-10,10),random(-16,16),random(25,65),0,0,frandom(1,3));
	A_SpawnItemEx("PM_OrangeEmbersFloat",0,0,random(25,65),FRandom(2,-2),FRandom(2,-2),FRandom(2,-2),0,128,220);
	}
	NULL A 0 A_StartSound("skeleton/step",5);
	TNT1 A 0 A_CheckSight("See");
	Loop;
  Melee:
	TNT1 A 0 A_Recoil(-8);
	R10_ G 5
	{
		A_FaceTarget();
		A_Pain();
	}
	TNT1 A 0
	{
		A_StartSound("GenericChargeHeavy", 15);
		A_StartSound("Tyrant/ShellLoadReady", 16, pitch: 1.4);
	}
	R10_ EEEEEEE 3 A_FaceTarget();
	R10_ E 0
	{
		A_StartSound("ExileFireExplode", 15, pitch: 1.2);
		A_StartSound("PMInfernal/Explode", 16, pitch: 1.65);
	}
	R10_ FF 3
	{
		A_FaceTarget();
		A_SpawnItemEx("OrangeShockwaveBig", zofs: 46);
		for (int i = random(4, 6); i > 0; i--) A_SpawnItemEx("PM_HuskFireGlobs", 0, 0, 46, frandom(4, 8), 0, frandom(-4, 4), frandom(-40, 40), flags: SXF_NOCHECKPOSITION);
		for (int i = random(2, 4); i > 0; i--) A_SpawnItemEx("PM_FireballExplosionFlames", 0, 0, 46, frandom(5, 10), 0, frandom(-5, 5), frandom(-40, 40), flags: SXF_NOCHECKPOSITION);
		
		if (target && checksight(target) && distance3D(target) < 165)
		{
			A_Explode(25,165,XF_NOTMISSILE,0,50,0,0,"BulletPuff","GhoulCloud");
			target.Vel3DFromAngle(15, self.angle, self.pitch);
			target.vel.z += 7;
		}
	}
	R10_ EEAA 5 A_FaceTarget();
    Goto See;
  Missile:
	TNT1 A 0;
	TNT1 A 0 A_Jump(100, "MissileStrike");
  HeavyMissile:
	TNT1 A 0
	{
		if (!random(0, 1))
			A_StartSound("Militant/Attack");
	}
	TNT1 A 0
	{
		A_StartSound("GenericChargeHeavy", 15);
		A_StartSound("Tyrant/ShellLoad", 16, pitch: 1.4);
	}
	R10_ EEEEEEEE 3 A_FaceTarget();
	R10_ E 0
	{
		A_StartSound("ExileFireExplode", 15, pitch: 1.2);
		A_StartSound("PMInfernal/Explode", 16, pitch: 1.65);
	}
	R10_ F 3
	{
		A_FaceTarget();
		A_SpawnItemEx("OrangeShockwaveBig", zofs: 60);
		A_SpawnProjectile("PM_MilitantHeavyRocket", 46);
	}
	R10_ EEEAAA 5 A_FaceTarget();
    Goto See;
  MissileStrike:
	TNT1 A 0
	{
		if (!random(0, 1))
			A_StartSound("Militant/Attack");
	}
	TNT1 A 0 { pm_attackloop = random(0, 2); }
	TNT1 A 0 A_StartSound("Director/ASPodExtent",20,volume:0.7,pitch:1.6);
	R10_ GGGGGGGGG 2 A_FaceTarget(360,180);
  MStrikeLoop:
	R10_ H 5 Bright Light("AvatarFireball2")
	{
		A_StartSound("Director/AStrikeShot",20,CHANF_OVERLAP,pitch:1.5);
		A_SpawnProjectile("PM_KurganMissileStrike", 0, -13, 86, CMF_AIMDIRECTION|CMF_OFFSETPITCH, frandom(-40,-60));
		A_FaceTarget(4,4);
	}
	R10_ G 5 Bright
	{
		if(pm_attackloop >= 6)
		{ return ResolveState("MStrikeEnd"); }
	    A_FaceTarget(3,3);
		pm_attackloop++;
		return ResolveState(null);
	}
	Loop;
  MStrikeEnd:
	TNT1 A 0 A_StopSound(36);
	R10_ GGGGGG 2 A_FaceTarget(3,3,0,0);
	Goto See;
  Pain:
    R10_ I 5;
    R10_ I 5 A_Pain;
    Goto See;
  Death:
	TNT1 A 0 PM_Scream();
    R10_ IJKLM 5;
    R10_ N -1;
    Stop;
  Raise:
    R10_ Q 5;
    R10_ PONML 5;
    Goto Look;
  }
}

		}

		A_SeekerMissile(1,2,SMF_LOOK);
		}
		Loop;
	Death:
		TNT1 A 0 
		{
		A_Explode(40,250);
		Radius_Quake (2, 34, 0, 10, 0);
		}
		TNT1 A 0 A_RadiusThrust(9000,90);
		TNT1 A 0 A_StartSound("BigExplosionDistant", CHAN_7);
		TNT1 A 0 A_SpawnItemEx("OrangeShockwaveBig");
		TNT1 A 0 A_SpawnItemEx("OrangeExplosionFlare");
		//TNT1 A 0 A_SpawnItemEx("OrangeEmbersSpawn");

		//TNT1 A 0 A_SpawnItemEx("OFireBallExplode");
		TNT1 AAA 0 A_SpawnItemEx("PM_ExileFireShrapnel",0,0,0,random(-2,15),random(-2,15),random(-2,15),0,SXF_NOCHECKPOSITION,70);
		TNT1 AAAA 0 A_SpawnItemEx("PM_ExplosionLongerSmokeBig",0,0,1,random(-8,8),random(-8,8),random(0,3),0,0,70);
