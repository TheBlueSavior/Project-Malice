Class PM_ApostateScanner : PMGreedMonsterBase
{
	Default
	{
		PMMonsterBase.CodexLore
		"Those drones are scanners originally created for city inspection and crime protection, only for those to be turned into hellish scanners used against oppressors during and after the invasion of Earth.
		Those who do not comply to the scanner perish. Once nearby, they will start to spew poison gas in front of you until you go far away from its range, or even blind your vision for a short time.";
		Health 85;
		PainChance 255;
		Speed 3;
		Radius 16;
		Height 25;
		Mass 100;
		Monster;
		SeeSound  "Furor/Sight";
		PainSound "Furor/Pain";
		DeathSound "Furor/Death";
		ActiveSound "Furor/Sight";
		AttackSound "Furor/Scream";
		Renderstyle "Normal";
		DamageFunction 15;
		DeathHeight 4;
		Obituary "%o couldn't comply to the Apostate Scanner.";
		Tag "Apostate Scanner";
		Species "Soul";
		Scale 1.2;
		//+NOINFIGHTING
		+FLOATBOB
		+FLOAT
		FloatBobStrength 0.2;
		+NOGRAVITY
		+NOICEDEATH
		+DONTFALL
		+FORCEXYBILLBOARD
		+DONTHARMCLASS
		+DONTHARMSPECIES
		+MISSILEEVENMORE
		+NOBLOODDECALS
		Bloodtype "PM_NormalHitSpark";
		TeleFogSourceType "PM_SmallDemonicTeleportFog";
		TeleFogDestType "PM_SmallDemonicTeleportFog";
	}
	
	bool loopplay;
	
	void StartHumming(bool act = true)
	{
		if (act)
		{
			if (!loopplay)
			{
				A_StartSound("ApostateScanner/Loop", 80, CHANF_LOOPING);
				loopplay = true;
			}
		}
		else
		{
			loopplay = false;
			A_StopSound(80);
		}
	}
	
	override void tick()
	{
		super.Tick();
		if (!level.isFrozen())
			vel *= 0.96;
	}
	
	States
	{
	Spawn:
		O03_ A 10 PM_Look();
		Loop;
	Look:
		O03_ A 0 StartHumming();
		O03_ A random(1, 5)
		{
			PM_LookingForPlayer();
			A_Recoil(-0.25);
		}
		TNT1 A 0 A_JumpIfTargetInLOS("Spotted",270,JLOSF_DEADNOJUMP);
		Loop;
	Spotted:
		TNT1 A 0 PM_AlertSound();
		Goto See;
	See:
		O03_ A random(1, 5)
		{
			A_Chase("Melee",null);
			if (target && distance3D(target) < 200) A_FaceTarget(20,20);
			else A_FaceTarget(2,1);
			if (target && distance3D(target) < 85) A_Recoil(0.25);
			else A_Recoil(-0.25);
		}
		Loop;
	Pain:
		O03_ D 2;
		O03_ D 2 A_Pain();
		Goto See;
	Melee:
		Goto See;
	Death:
		TNT1 A 0;
		TNT1 A 0
		{
			StartHumming(false);
			A_ScreamAndUnblock();
		}
		O03_ D 1;
		Stop;
	}
}
