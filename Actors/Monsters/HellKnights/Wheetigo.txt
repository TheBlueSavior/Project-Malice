Class PM_Wheetigo : PMFamineMonsterBase
{
	Default
	{
		PMMonsterBase.CodexLore
		"It is batshit insane.
		
		These skinwalker-looking abominations are driven to extreme madness due to famine, and for some reason it has magical powers?
		It used to be an archdemon residing within the apocalypse of Famine, but within time, they have started to become unhinged, letting their lust for hunger take over.
		Famine truly is wicked and sick.
		
		Has ludircious speeds, will slink you into a frenzy once in contact with its prey. Its screeches can be very damaging.";
		Health 600;
		Radius 20;
		Height 56;
		Speed 18;
		PainChance 80;
		Mass 400;
		+FLOORCLIP
		+HARMFRIENDS
		+DOHARMSPECIES
		+QUICKTORETALIATE
		+SLIDESONWALLS
		+MISSILEMORE
		+MISSILEEVENMORE
		Monster;
		SeeSound "PMWheetigo/Sight";
		PainSound "PMWheetigo/Pain";
		DeathSound "PMWheetigo/Death";
		ActiveSound "PMWheetigo/Active";
		Tag "Wheetigo";
		Obituary "%o thought %h saw a skinwalker.";
		HitObituary "STOP IT";
		Species "Nobles";
		Bloodtype "PM_DarkRedBloodBase";
		TeleFogSourceType "PM_SmallDemonicTeleportFog";
		TeleFogDestType "PM_DemonicTeleportFog";
		MeleeThreshold 120;
		MaxTargetRange 1250;
	}
	
	override void Tick()
	{
		super.Tick();
		HandleEating(600);
	}
	
	void A_WheetigoThrow(class<actor> mymissile = "PM_WheetigoBomb", double ballang = 0)
	{
		int arc;
		let p = A_SpawnProjectile(mymissile,40,0,ballang);
		if(distance2d(target) <= 300) arc = 20;
		else arc = 75;
		if(p) p.vel = CalcThrowVector(pos, target.pos, p.gravity, arc);
		A_FaceTarget();
	}
   
	States
	{
	Spawn:
		F_HK H 0 nodelay { bSPRITEFLIP = false; eaten = false; }
	Idle:
		F_HK H 10 PM_Look();
		Loop;
	Look:
		F_HK H random(1,100);
		F_HK H 10 A_JumpIfTargetInLOS("Spotted",270,JLOSF_DEADNOJUMP);
		Loop;
	Spotted:
		TNT1 A 0 A_Jump(128, "SCREEEEEEEEEEEEEEEE");
		Goto Roar;
	Roar:
		F_HK H 0 PM_AlertSound();
		Goto See;
	SCREEEEEEEEEEEEEEEE:
		F_HK E 1 A_StartSound("PMWheetigo/SCREECH", CHAN_VOICE);
		F_HK EEEEEEEEEEEEEEEEEEEEEEEEEEEE random(1, 2)
		{
			A_FaceTarget();
			
			BlockThingsIterator ti = BlockThingsIterator.Create(self, 350);
			actor a;
			
			while(ti.Next())
			{
				a = ti.thing;
				double rad = abs((350 - Distance3D(a)) / 2);
				if (a && a.Health > 0 && a.bSHOOTABLE && !a.bINVULNERABLE && !a.bDORMANT && !a.bNODAMAGE && CheckSight(a) && distance3D(a) < 350)
				{
					if (!(a is "PM_Wheetigo"))
					{
						if (a.FindState("Pain"))
							a.SetStateLabel("Pain");
						a.DamageMobj(self, self, (a is "PlayerPawn") ? 1 : random(5, 15), "FunniScream", DMG_THRUSTLESS);
					}
				}
			}
			
			A_QuakeEx(5, 5, 0, 8, 0, 800, "", QF_3D|QF_SCALEDOWN);
		}
		F_HK E 0 A_StopSound(CHAN_VOICE);
		Goto See;
	See:
		F_HK AABBCCDD 2
		{
			bFRIGHTENED = FALSE;
			A_Recoil(-0.65);
			A_Chase();
		}
		TNT1 A 0 A_Jump(40, "Rush");
		Loop;
	Rush:
		F_HK ABCD 1
		{
			A_Recoil(-1);
			A_SpawnItemEx("PM_GenericPhaseGhost", zofs:6, flags: SXF_TRANSFERSPRITEFRAME|SXF_TRANSFERSCALE|SXF_TRANSFERRENDERSTYLE|SXF_TRANSFERSTENCILCOL);
			for (int i = random(2,3); i > 0; i--) A_Chase("Bomb", null);
			if (target && distance3D(target) < 80) SetStateLabel("Lunge");
		}
		TNT1 A 0 A_Jump(10, "See");
		Loop;
	Pain:
		F_HK H 2 { bSPRITEFLIP = false; }
		F_HK H 2 A_Pain();
		TNT1 A 0 A_Jump(80, "Rush");
		Goto See;
	Bomb:
		TNT1 A 0 A_StartSound("PMWheetigo/Attack", CHAN_VOICE);
		F_HK EEEEEEEEEEEEEEE 1
		{
			A_FaceTarget();
			A_SpawnItemEx("PM_GenericPhaseGhost", zofs:6, zvel:4, flags: SXF_TRANSFERSPRITEFRAME|SXF_TRANSFERSCALE|SXF_TRANSFERRENDERSTYLE|SXF_TRANSFERSTENCILCOL);
		}
		F_HK FG 4 A_FaceTarget();
		F_HK G 15 A_WheetigoThrow();
		Goto See;
	Missile:
		TNT1 A 0 A_Jump(80, "Bomb","Lunge");
		TNT1 A 0 A_Jump(5, "SCREEEEEEEEEEEEEEEE");
	Leap:
	    TNT1 A 0 
	    { 
	     if(target && Distance3D(target) >= 768)
	      { SetStateLabel("Rush"); }
	    }
		TNT1 A 0 A_StartSound("PMWheetigo/Sight",13);
		F_HK HHE 5 A_FaceTarget();
		F_HK E 2 PM_LeapTo(target, 1200, "Rush");
		TNT1 A 0 A_StartSound("Squire/Leap",15);
		TNT1 A 0 A_Recoil(-6);
	LeapLoop:
		F_HK EEEE 1 
		{
		 if(random(1,15) == 1)
		  { ThrustThing(random(0,360)*256/360,3,0,0); }
		  A_SpawnItemEx("PM_GenericPhaseGhost", zofs:6, flags: SXF_TRANSFERSPRITEFRAME|SXF_TRANSFERSCALE|SXF_TRANSFERRENDERSTYLE|SXF_TRANSFERSTENCILCOL);
		}
        TNT1 A 0 A_CheckFloor("LeapEnd");
		Loop;
	LeapEnd:
		 TNT1 A 0 A_StartSound("Doom4/Monsters/BaronFloorPound",CHAN_7,CHANF_OVERLAP,pitch:1.65);
		 TNT1 A 0 A_Quake(4,20,0,512);
		 TNT1 AAAAAAAAAAAAAAAA 0 A_SpawnItemEx("PM_BulletShrapnelNoCheck",random(-5,5),random(-5,5),0,random(-3,3),random(-5,5),random(5,20),0,SXF_NOCHECKPOSITION);
         TNT1 AAAAA 0 A_SpawnItemEx("PM_BoomSmoke",random(-1,1),random(-1,1),random(2,4),random(-1,1),random(-1,1),random(0,1),0,SXF_NOCHECKPOSITION);
         TNT1 A 0 { if(target && CheckSight(target) && Distance3D(target) < 145) SetStateLabel("DoDamage"); }
		 Goto ResumeLeapEnd;
	Lunge:
		F_HK H 20 A_FaceTarget();
		F_HK G 1
		{
			A_FaceTarget();
			A_StartSound("PMWheetigo/SCREECH", CHAN_VOICE, attenuation: 2.2);
		}
		F_HK EEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEE 1
		{
			let fx = Spawn("PM_GreenFlameTrails", Vec3Offset(0,-18,60));
			if (fx)
			{
				fx.A_SetRenderStyle(1, STYLE_ADDSHADED);
				fx.SetShade("8f807c");
				fx.vel.z += 4;
			}
				
			A_Recoil(-5);
			A_SpawnItemEx("PM_GenericPhaseGhost", zofs:6, flags: SXF_TRANSFERSPRITEFRAME|SXF_TRANSFERSCALE|SXF_TRANSFERRENDERSTYLE|SXF_TRANSFERSTENCILCOL);
			if(target && CheckSight(target) && Distance3D(target) < 90) SetStateLabel("Splode");
		}
		Goto See;
	Splode:
		F_HK E 30
		{
			A_StopSound(CHAN_VOICE);
			A_StartSound("Famine/Explode",CHAN_5,volume:0.85,pitch:frandom(0.9,1.3));
			A_StartSound("Noble/BombExp",CHAN_6,pitch:1.3);
			A_StartSound("Noble/BombExpFar",CHAN_7,pitch:1.3);
			A_Recoil(10);
			vel.z += 10;
			A_Explode((40),200);
			A_QuakeEx(5,5,5,10,0,400,0,0,0,0,0,150);
			
			for (int i = 6; i > 0; i--)
			{
				let fx1 = Spawn("PM_GenericExplosionFlames", Vec3Offset(0,0,0));
				if (fx1)
				{
					fx1.A_SetRenderStyle(1, STYLE_ADDSHADED);
					fx1.SetShade("8f807c");
				}
			}
			
			BlockThingsIterator ti = BlockThingsIterator.Create(self, 350);
			actor a;
			
			while(ti.Next())
			{
				a = ti.thing;
				double rad = abs((350 - Distance3D(a)) / 2);
				if (a && a.Health > 0 && a.bSHOOTABLE && !a.bINVULNERABLE && !a.bDORMANT && !a.bNODAMAGE && CheckSight(a) && distance3D(a) < 999 && !a.bDONTTHRUST)
				{
					a.Vel3DFromAngle((rad / 8) / max(1, a.Mass / 2), self.AngleTo(a), self.PitchTo(a));
					if (a.GetMaxHealth() < 2500)
						a.vel.z += (rad / 16);
					a.vel.z = clamp(a.vel.z, -25, 25);
				}
			}
		}
		TNT1 A 0 A_QuakeEx(5,5,5,10,0,400,0,0,0,0,0,150);
		TNT1 A 8 Bright A_Explode((40),200);
		Goto See;
	DoDamage:
		 TNT1 A 0 A_BetaSkullAttack;
		 Goto ResumeLeapEnd;
	ResumeLeapEnd:
		 F_HK G 1 A_Stop;
		 F_HK G 6;
		 Goto See; 
	Death:
		F_HK H 15 { bSPRITEFLIP = false; }
		F_HD A 5 A_ScreamAndUnblock();
		F_HD B 4 A_SpawnItemEx("PM_DarkRedMeatChunkLauncher",random(-2,2),random(-2,2),random(20,50),0,0,0,0,SXF_NOCHECKPOSITION);
		F_HD CDEFGH 4 A_SpawnItemEx("PM_BlackInstestinGib",random(-4,4),random(-4,4),random(25,35),random(-3,3),random(-3,3),random(0,5),random(-180,180),SXF_NOCHECKPOSITION);
		F_HD H -1;
		Stop;
	Melee:
		TNT1 A 0 A_Jump(60, "SCREEEEEEEEEEEEEEEE");
		F_HK E 10 A_StartSound("PMWheetigo/Attack", CHAN_VOICE);
		F_HK F 2 A_FaceTarget();
		F_HK G 2
		{
			A_StartSound("Doom4/Other/MeleeSwing",22,CHANF_OVERLAP,pitch:1.75);
			A_Recoil(-7);
			A_SpawnProjectile("PM_MeleeHitbox",flags:CMF_AIMDIRECTION,pitch);
		}
		F_HK F 0 { bSPRITEFLIP = true; }
		F_HK EF 2 A_FaceTarget();
		F_HK G 2
		{
			A_StartSound("Doom4/Other/MeleeSwing",22,CHANF_OVERLAP,pitch:1.75);
			A_Recoil(-7);
			A_SpawnProjectile("PM_MeleeHitbox",flags:CMF_AIMDIRECTION,pitch);
		}
		F_HK F 0 { bSPRITEFLIP = false; }
		F_HK E 2 A_FaceTarget();
		F_HK F 0 { if (target && Distance3D(target) > 120) SetStateLabel("See"); }
		Goto Melee+2;
	Eat:
		F_HK H 20;
		F_HK A 0 { eaten = false; bNOPAIN = false; }
		Goto See;
	Raise:
		F_HD GFEFCBA 7;
		Goto See;
	}
}

  {
  Spawn:

		{
		A_SpawnItemEx("PM_GreenFlameTrails",0,frandom(-6,6),frandom(-6,6),frandom(-2,0),flags:SXF_TRANSFERTRANSLATION);

  Death:

	TNT1 A 0 A_QuakeEx(5,5,5,10,0,400,0,0,0,0,0,150);
	TNT1 A 8 Bright A_Explode(35,200);
	Stop;
	}
}

Class PM_MiniFamineBomb : PM_WheetigoBomb
{
  Default
  {
  Speed 10;
  Scale 0.25;
  Gravity 1;
  Bouncetype "Doom";
  BounceFactor 0.9;
  DamageFunction (7);
  WallBounceFactor 1.2;
  BounceCount 3;
  Translation "0:255=%[0.00,0.00,0.00]:[1.13,0.96,0.88]";
  }
  States
  {
  Spawn:
	TNT1 A 0 NoDelay { user_grenadetime = random(0,20); }
	SPKO AAAAAA 1 Bright
		{
		A_SpawnItemEx("PM_GreenFlameTrails",0,frandom(-6,6),frandom(-6,6),frandom(-2,0),flags:SXF_TRANSFERTRANSLATION|SXF_TRANSFERSCALE);
		A_SpawnItemEx("PM_DevilBallSmoke",0,frandom(-2,2),frandom(-2,2),0,0,frandom(0,2),flags:SXF_TRANSFERSCALE);
		user_grenadetime++;
		roll = random(0, 360);
		if (waterlevel >= 2)
			{
			A_SetGravity(4.0);
			A_SetMass(1500);
			}
		}
	Goto Spawn+1;
  Bounce:
	TNT1 A 0
	{
		if (user_grenadetime >= 40) SetStateLabel("Death");
	}
	Goto See;
  Death:
  XDeath:
	TNT1 A 0 A_StartSound("Nymphite/BallPop",CHAN_7,pitch:frandom(0.7, 1.1));
	TNT1 A 0 A_StartSound("Noble/BombExp",CHAN_6,pitch:1.7);
	TNT1 A 0 { bNOGRAVITY = true; }
	TNT1 AAA 0 A_SpawnItemEx("PM_FireGlobs",0,0,0,random(-3,3),random(-3,3),random(-3,3),0,SXF_NOCHECKPOSITION|SXF_TRANSFERTRANSLATION);
	TNT1 AAA 0 A_SpawnItemEx("PM_SmallerGenericExplosionFlames",0,0,0,random(-3,3),random(-3,3),random(-3,3),0,SXF_NOCHECKPOSITION|SXF_TRANSFERTRANSLATION);
	TNT1 A 0 A_QuakeEx(2,2,2,10,0,400,0,0,0,0,0,150);
	TNT1 A 8 Bright A_Explode(20,100);
	Stop;
	}
