 override void Tick()
 {
	super.Tick();
	if (health <= 0)
	{
		A_StopSound(30);
		A_StopSound(31);
		return;
	}
	if (target && distance3D(target) <= 1800) A_StartSound("PMCataphract/Engine", 30, CHANF_LOOPING);
	else A_StopSound(30);

	if (--exhaustTimer <= 0)
	{
		exhaustTimer = random(2, 4);
		let cs = CurState;
		if (cs && !InStateSequence(cs, FindState("Spawn")) && !InStateSequence(cs, FindState("Look")))
		{
			A_SpawnParticleEx("555555", TexMan.CheckForTexture("GNSML0"),
				style: STYLE_Normal, flags: SPF_RELATIVE | SPF_ROLL,
				lifetime: random(50, 80), size: frandom(8, 16),
				xoff: -28, yoff: 0, zoff: 50,
				velx: frandom(-0.8, 0.3), vely: frandom(-0.5, 0.5), velz: frandom(2.5, 4.0),
				accelz: -0.03,
				startalphaf: frandom(0.4, 0.65), fadestepf: -1,
				sizestep: frandom(0.5, 1.0), startroll: random(-180, 180), rollvel: frandom(-1.2, 1.2));
		}
	}
 }

 void CataAimLaser(double side = 0)

	if (!target) return;
	FLineTraceData lt;
	LineTrace(angle, 2200, pitch, 0, height * 0.6, data: lt);
	double sa = sin(angle);
	double ca = cos(angle);
	double fwd = 24;
	Vector2 relOff = (fwd * ca + side * sa, fwd * sa - side * ca);
	PM_SpawnBolt(self, lt.HitLocation, "ff0000", TexMan.CheckForTexture("RLSSA0"), maxDiff: 0, scale: 0.25, offset: (relOff.X, relOff.Y, 32), baseTime: 2, fadeTime: 0);
